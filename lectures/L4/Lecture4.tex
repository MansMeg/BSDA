\begin{frame}
\frametitle{Riemann recap}

  Partition $[a,b]$ as
  \[
    a=x_0<x_1<\cdots<x_S=b,
    \qquad \Delta x_s=x_s-x_{s-1},
  \]
  and choose one point $x_s^*\in[x_{s-1},x_s]$ in each subinterval.
  The corresponding \uured{Riemann sum} is
  \[
    R_S=\sum_{s=1}^{S} f(x_s^*)\,\Delta x_s.
  \]
  \pause
  The \uured{Riemann integral} is the limit
  \[
    \int_a^b f(x)\,dx
    =\lim_{\max_s \Delta x_s\to 0}
      \sum_{s=1}^{S} f(x_s^*)\,\Delta x_s.
  \]


\end{frame}

\begin{frame}
\frametitle{Numerical integration}


  \begin{itemize}
    \item Use a finite Riemann sum to approximate the integral:
    \pause
    \begin{align*}
         {\displaystyle \int _{a}^{b}f(x)\,dx \approx \sum _{s=1}^{S}f(x_{s}^{*})\,\Delta x_{s},}
    \end{align*}
    \item For $S$ equal-width subintervals,
      $\Delta x=(b-a)/S$.
    \item A common choice is the \uured{midpoint}:
    \[
      x_s^* = \frac{x_{s-1}+x_s}{2}
      = a + \left(s-\tfrac{1}{2}\right)\Delta x,
      \qquad s=1,\ldots,S.
    \]

  \end{itemize}

\end{frame}

\begin{frame}
\frametitle{Monte Carlo integration: recap}

  \begin{itemize}
    \item From Lecture 3, for i.i.d.\ draws
      $\theta^{(1)},\ldots,\theta^{(S)}\sim p(\theta\mid y)$,
      \[
        I_h=\E[h(\theta)\mid y]
        =\int h(\theta)p(\theta\mid y)\,d\theta.
      \]
    \pause
    \item The Monte Carlo estimator is
      \[
        \hat I_S=\frac{1}{S}\sum_{s=1}^S h(\theta^{(s)}).
      \]
    \item<3-> Unlike a grid, its evaluation points are random draws from
      the distribution inside the integral.
  \end{itemize}

\end{frame}

\begin{frame}
\frametitle{Unbiasedness and MCSE}

  If $\E[|h(\theta)|\mid y]<\infty$, then $\hat I_S$ is \uured{unbiased}:
  \[
    \E[\hat I_S\mid y]
    =\frac{1}{S}\sum_{s=1}^S \E[h(\theta^{(s)})\mid y]
    =I_h.
  \]
  \pause
  If $\sigma_h^2=\Var[h(\theta)\mid y]<\infty$, independence gives
  \[
    \Var(\hat I_S\mid y)=\frac{\sigma_h^2}{S},
    \qquad
    \operatorname{MCSE}(\hat I_S)=\frac{\sigma_h}{\sqrt S}.
  \]
  \pause
  Estimate MCSE by replacing $\sigma_h$ with the sample standard deviation
  of $h(\theta^{(1)}),\ldots,h(\theta^{(S)})$.

\end{frame}

\begin{frame}
\frametitle{Monte Carlo integration: convergence}

\begin{itemize}
  \item \uured{Strong law of large numbers}: if
    $\E[|h(\theta)|\mid y]<\infty$, then
    \[
      \hat I_S \xrightarrow{\mathrm{a.s.}} I_h.
    \]
  \item<2-> \uured{Mean-square convergence}: if $\sigma_h^2<\infty$, then
    \[
      \E[(\hat I_S-I_h)^2\mid y]=\frac{\sigma_h^2}{S}\longrightarrow 0.
    \]
  \item<3-> \uured{Central limit theorem}: if $0<\sigma_h^2<\infty$, then
    \[
      \sqrt S\,(\hat I_S-I_h)
      \xrightarrow{d}\N(0,\sigma_h^2).
    \]
  \item<4-> Thus the typical Monte Carlo error decreases at rate
    $S^{-1/2}$: halving MCSE requires four times as many draws.
\end{itemize}

\end{frame}

\begin{frame}


\frametitle{Monte Carlo vs. deterministic integration}

  \begin{itemize}
  \item In one dimension, for a sufficiently smooth function, typical
    error rates with $N$ evaluations are
    \[
      \begin{array}{c|c}
        \text{Method} & \text{Error rate} \\
        \hline
        \text{Midpoint or trapezoidal rule} & N^{-2} \\
        \text{Simpson's rule} & N^{-4} \\
        \text{Monte Carlo} & N^{-1/2}
      \end{array}
    \]
  \item Deterministic integration is therefore usually preferable for
    a smooth, low-dimensional problem.
  \only<1>{\\{\color{uured}Why do we still need Monte Carlo?}}
  \pause
  \item A grid with $m$ points in each of $D$ dimensions requires
    $N=m^D$ evaluations: its cost grows exponentially with $D$.
  \item<3-> The Monte Carlo rate $N^{-1/2}$ does not explicitly depend on
    $D$, which makes Monte Carlo attractive in higher dimensions.
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Posterior uncertainty vs. Monte Carlo error}

  \begin{itemize}
  \item \uured{Posterior uncertainty} is the spread of $p(\theta\mid y)$
    \begin{itemize}
      \item it describes what we know about $\theta$ after observing the data
      \item it does not disappear when we generate more posterior draws
    \end{itemize}
  \pause
  \item \uured{Monte Carlo error} comes from using finitely many draws
    to estimate a posterior summary
    \[
      \widehat{\E(\theta\mid y)}=\frac{1}{S}\sum_{s=1}^S\theta^{(s)},
      \qquad
      \operatorname{MCSE}=\frac{\sigma_\theta}{\sqrt S}.
    \]
    \begin{itemize}
      \item it decreases when we generate more draws
    \end{itemize}
  \pause
  \item These are different sources of uncertainty and should be
    assessed separately.
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Estimating a posterior probability}

  \begin{itemize}
  \item Let $p_A=P(\theta\in A\mid y)$ and define
    $I_s=I(\theta^{(s)}\in A)$. Then
    \begin{equation*}
      I_s\sim\operatorname{Bernoulli}(p_A),
      \qquad
      \widehat p_A=\frac{1}{S}\sum_{s=1}^S I_s.
    \end{equation*}
    \pause
  \item The number of draws in $A$ is
    $\sum_s I_s\sim\operatorname{Binomial}(S,p_A)$, so
    \[
      \operatorname{MCSE}(\widehat p_A)
      =\sqrt{\frac{p_A(1-p_A)}{S}}
      \approx\sqrt{\frac{\widehat p_A(1-\widehat p_A)}{S}}.
    \]
    \pause
  \item If $S=100$ and $p_A\approx0.5$, MCSE is $0.05$, or about
    5 percentage points. An MCSE of $0.01$ requires about 2500 draws.
    \pause
  \item Small probabilities require many draws
    \begin{itemize}
    \item the expected number of draws in $A$ is $S p_A$, so we need
      $S$ much larger than $1/p_A$ for a stable estimate
    \end{itemize}
\end{itemize}

\end{frame}

    \begin{itemize}
    \item deterministic methods
    \item Rao-Blackwellization (marginalisation)
    \item variance-reduction methods, such as control variates
    \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{How many simulation draws are needed?}

  \begin{itemize}
  \item The $S^{-1/2}$ Monte Carlo rate does not explicitly depend on dimension
\pause
    \item However, obtaining useful independent draws can be difficult in
      high-dimensional problems

  \end{itemize}

\end{frame}


\section{Direct sampling}
\frame{\sectionpage}

\begin{frame}
\frametitle{Direct sampling}

  \begin{itemize}
  \item Direct simulation from a known density or mass function, such as
    $p(\theta\mid y)$ in a conjugate model
  \item Produces independent draws
    \begin{itemize}
    \item analytic transformations of uniform random numbers
      (e.g. Appendix A)
    \item factorization
    \item numerical inversion of the CDF
    \end{itemize}
  \item {\color{uured}Limitation}: available only for a limited set of models
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Random number generators}

  \begin{itemize}
  \item {\color{uured}How do we sample from a probability distribution?}
  \pause
  \item Good {\color{uured}pseudorandom} number generators are sufficient
    for Bayesian inference
  \begin{itemize}
    \item a pseudorandom generator uses a deterministic algorithm to
      produce a sequence that behaves like a random sequence
    \item modern statistical software uses high-quality generators
    \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}


\frametitle{Direct simulation: example}

  \begin{itemize}
  \item Box--Muller method:\\ If $U_1$ and $U_2$ are independent
    draws from $\U(0,1)$, and
    \begin{align*}
      X_1 & = \sqrt{-2\log(U_1)}\cos(2\pi U_2) \\
      X_2 & = \sqrt{-2\log(U_1)}\sin(2\pi U_2)
    \end{align*}
    then $X_1$ and $X_2$ are independent draws from the distribution
    $\N(0,1)$
    \pause
    \begin{itemize}
      \item it is not the fastest method because it requires trigonometric functions
      \item many alternative methods exist for the normal distribution
      \item R uses numerical inversion by default
    \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Effective sample size}

  \begin{itemize}
    \item The effective sample size $S_\text{eff}$ compares Monte Carlo
      precision with that of independent, equally weighted draws
    \item For $S$ independent draws, $S_\text{eff}=S$
    \pause
    \item Weighted or correlated draws usually have $S_\text{eff}<S$
    \item A larger $S_\text{eff}$ means more precise estimates for the
      same number of stored draws
  \end{itemize}

\end{frame}

\subsection{Rejection sampling}
\frame{\subsectionpage}

\begin{frame}

\frametitle{Rejection sampling}

    \vspace{-.3\baselineskip}
  \begin{itemize}
  \item[-] Let $p(\theta)=q(\theta)/Z$ be the target density
  \item[-] Choose a proposal $g$ and a finite $M$ such that
    \[
      q(\theta)\leq M g(\theta) \quad \text{for all }\theta
    \]
  \item[-] Intuitively, scale $g$ vertically by $M$ until $Mg$ covers
    $q$ everywhere. Then $q(\theta)/(Mg(\theta))$ is always between 0 and 1.
  \item<2->[-] Draw $\theta^*\sim g$ and $u\sim\U(0,1)$; accept if
    \[
      u\leq\frac{q(\theta^*)}{M g(\theta^*)}
    \]
  \end{itemize}

  \begin{center}
    \vspace{-1.6\baselineskip}
    \only<1>{\includegraphics[width=6cm]{figs/rejection1.pdf}}
    \only<2>{\includegraphics[width=6cm]{figs/rejection2.pdf}}
    \only<3->{\includegraphics[width=6cm]{figs/rejection3.pdf}}
  \end{center}

\end{frame}

\begin{frame}
\frametitle{Why rejection sampling works}

\begin{itemize}
  \item The density of an accepted proposal is proportional to
    \[
      \underbrace{g(\theta)}_{\text{proposal density}}
      \underbrace{\frac{q(\theta)}{M g(\theta)}}_{\text{acceptance probability}}
      =\frac{q(\theta)}{M}\propto p(\theta).
    \]
  \item<2-> Thus the accepted values are independent draws from the target
  \item<3-> If $A$ of $T$ proposals are accepted, their effective sample
    size is $A$, while $A/T$ estimates the acceptance rate $Z/M$
  \only<3>{\\{\color{uured} When will the acceptance rate be high or low?}}
  \pause
    \begin{itemize}
    \item<4-> A tight envelope gives a high acceptance rate
    \item<4-> Finding a good envelope becomes difficult as dimension increases
    \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}{Pair discussion: rejection proposal}

  \begin{itemize}
  \item What happens if $Mg$ fails to cover the target everywhere?
  \item What happens if the envelope is valid but much wider than the target?
  \end{itemize}

  \uncover<2->{
  \vspace{0.5\baselineskip}
  {\small
  {\color{uured}After discussion:}
  \begin{itemize}
  \item Without full coverage, the algorithm is invalid and misses part of
    the target.
  \item An overly wide envelope is valid but gives a low acceptance rate.
  \item A good envelope covers the target as tightly as possible.
  \end{itemize}}}

\end{frame}

\subsection{Importance sampling}
\frame{\subsectionpage}

\begin{frame}
\frametitle{Importance sampling identity}

  If we can evaluate the normalized target $p$, then
  \begin{align*}
    w(\theta)&=\frac{p(\theta)}{g(\theta)},\\
    \E_p[f(\theta)]
    &=\int f(\theta)w(\theta)g(\theta)\,d\theta
      =\E_g[w(\theta)f(\theta)].
  \end{align*}
  \pause
  Thus, for $\theta^{(s)}\sim g$,
  \[
    \widehat I_S=\frac{1}{S}\sum_{s=1}^S
      w(\theta^{(s)})f(\theta^{(s)})
  \]
  is an unbiased estimator of $\E_p[f(\theta)]$.
  \pause
  If only $q(\theta)=Zp(\theta)$ is known, the unknown $Z$ cancels:
  \[
    \E_p[f(\theta)]
    =\frac{\E_g[(q/g)f]}{\E_g[q/g]}.
  \]

\end{frame}

\begin{frame}

\frametitle{Self-normalized importance sampling}

  \begin{itemize}
     \vspace{-.5\baselineskip}
   \item[-] Draw from $g$ and compute
     $w_s=q(\theta^{(s)})/g(\theta^{(s)})$
   \end{itemize}
   \begin{center}
     \vspace{-1\baselineskip}
     \only<1-2>{\includegraphics[width=9cm]{figs/importancesamp1.pdf}}
     \only<3>{\includegraphics[width=9cm]{figs/importancesamp2.pdf}}
     \vspace{-1\baselineskip}
     \only<2->{
   \begin{eqnarray*}
      \widehat{\E_p[f(\theta)]}
      = \sum_{s=1}^S \widetilde w_s f(\theta^{(s)}),
      \qquad
      \widetilde w_s=\frac{w_s}{\sum_{r=1}^S w_r}
   \end{eqnarray*}
   }
   \end{center}

   \only<3>{The estimator is generally biased for finite $S$, but
     consistent as $S\to\infty$.}

\end{frame}

\begin{frame}{Pair discussion: proposal tails}

  \begin{itemize}
  \item The target has heavier tails than the proposal.
  \item What happens to the importance weights in the tails?
  \item Would a slightly wider proposal be safer or less safe?
  \end{itemize}

  \uncover<2->{
  \vspace{0.5\baselineskip}
  {\small
  {\color{uured}After discussion:}
  \begin{itemize}
  \item Rare tail draws receive extremely large weights.
  \item A few draws may dominate, producing low ESS and unstable estimates.
  \item A proposal with sufficiently heavy tails is usually safer.
  \end{itemize}}}

\end{frame}

\begin{frame}

\frametitle{Choosing a proposal}

  \begin{itemize}
  \item The proposal must satisfy
    \[
      g(\theta)>0 \quad \text{wherever } p(\theta)>0
    \]
    Otherwise, important parts of the target can never be sampled.
    \pause
  \item A good proposal covers the target and gives similar weights
    \begin{itemize}
      \item tails that are too light can produce rare, extremely large weights
      \item large weight variation gives unstable estimates and low effective
        sample size
    \end{itemize}
    \pause
  \item Finding a good proposal becomes harder as dimension increases
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Weight variability}

  \begin{itemize}
  \item Variation in the weights affects the {\color{uured} effective sample size}
    \begin{itemize}
    \item if one weight dominates, we have effectively one draw
    \item if all weights are equal, we have effectively $S$ draws
    \end{itemize}
    \only<1>{{\color{uured}Which weight pattern would you trust more?}}
    \pause
  \item For the usual central limit theorem, the weighted quantity must
    have finite variance; for self-normalized importance sampling this
    requires
    \[
      \E_g\!\left[w(\theta)^2
      \bigl(f(\theta)-\E_p[f(\theta)]\bigr)^2\right]<\infty.
    \]
  \item<3-> Heavy-tailed weights can violate this condition even when
    $f(\theta)$ itself has finite variance under the target.
  \end{itemize}

\end{frame}

\begin{frame}
\frametitle{Bioassay importance-sampling workflow}

  \begin{enumerate}
    \item Draw $(\alpha^{(s)},\beta^{(s)})$ from the bivariate normal
      proposal $g$
    \item Compute log weights
      $\log w_s=\log q(\alpha^{(s)},\beta^{(s)})-
      \log g(\alpha^{(s)},\beta^{(s)})$
    \item Stabilize and normalize the weights so that
      $\sum_s\widetilde w_s=1$
    \item Compute weighted posterior summaries and inspect the weights
      and $S_{\rm eff}$
    \item Optionally resample draws with probabilities $\widetilde w_s$
  \end{enumerate}

  \pause
  \uured{IR} in the following figures means \uured{importance resampling}:
  resampling the proposal draws using the normalized weights.

\end{frame}

\subsection{Pareto-smoothed importance sampling}
\frame{\subsectionpage}

\begin{frame}

\frametitle{Pareto-smoothed importance sampling}

  \begin{itemize}
  \item Raw importance sampling can be dominated by a few very large weights
  \pause
  \item PSIS fits a generalized Pareto distribution to the upper tail
    of the importance ratios
  \item It replaces the largest ratios with smoothed values from the fitted tail
  \pause
  \item This usually reduces variability, at the cost of a small amount of bias
  \item PSIS also provides improved estimates of effective sample size and MCSE
  \end{itemize}

  \vfill
  {\tiny Vehtari, Simpson, Gelman, Yao, and Gabry (2024)}
\end{frame}

\begin{frame}

\frametitle{The Pareto $\widehat{k}$ diagnostic}

  \begin{itemize}
  \item $\widehat{k}$ describes the heaviness of the upper tail of the
    importance ratios
    \begin{itemize}
      \item smaller $\widehat{k}$ means more stable importance sampling
      \item larger $\widehat{k}$ means that a few ratios dominate
    \end{itemize}
  \pause
  \item Current guidance treats estimates as reliable when
    \[
      \widehat{k}<\min\!\left(1-\frac{1}{\log_{10}S},\,0.7\right)
    \]
  \item<3-> Above this threshold, the estimate and its MCSE may be unreliable;
    investigate the proposal or use a more robust computation
  \item<4-> For intuition: a generalized Pareto tail has finite variance
    when $k<1/2$; when $k\geq1$, even its mean does not exist
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Importance-sampling leave-one-out cross-validation}

  \begin{itemize}
  \item Later in the course, we use $p(\theta\mid y)$ as a proposal
    distribution for $p(\theta\mid y_{-i})$
    \begin{itemize}
    \item this allows fast approximation of leave-one-out cross-validation
      \begin{align*}
        p(y_i|y_{-i})=\int p(y_i|\theta) p(\theta|y_{-i}) d\theta
      \end{align*}
    \end{itemize}
  \end{itemize}

\end{frame}
