\section{Markov Chain Monte Carlo (MCMC)}
%\frame{\sectionpage}

\begin{frame}

\frametitle{Markov chains}

  \begin{itemize}
  \item<1-> A Markov chain is a sequence in which the next state depends
    on the current state, but not on earlier states
    \[
    p(\theta^{(t)}\mid\theta^{(t-1)},\ldots,\theta^{(0)})
    =p(\theta^{(t)}\mid\theta^{(t-1)}).
    \]
  \item<2-> $T(\theta^{(t)}\mid\theta^{(t-1)})$ is the
    {\color{uured}transition distribution}
  \item<3-> For a well-behaved chain, the distribution $P_t$ of
    $\theta^{(t)}$ approaches a stationary distribution $\pi$
  \item<4-> Total variation distance is
    \[
      d_{\mathrm{TV}}(P_t,\pi)
      =\sup_A\left|P_t(A)-\pi(A)\right|.
    \]
    It is the largest difference between the probabilities that $P_t$
    and $\pi$ assign to the same event $A$
  \item<5-> Convergence in total variation means
    $d_{\mathrm{TV}}(P_t,\pi)\rightarrow0$
  \item<6-> \uured{Goal in MCMC}: construct $T$ with
    $\pi(\theta)=p(\theta\mid y)$
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{A stationary distribution: two-state example}

  Consider a chain with states 1 and 2 and transition matrix
  \[
    T=
    \begin{pmatrix}
      0.8 & 0.2\\
      0.4 & 0.6
    \end{pmatrix}.
  \]
  Each row gives the probabilities of the next state.
  \pause
  \begin{itemize}
  \item The distribution $\pi=(2/3,1/3)$ is stationary because
    \[
      \pi T=(2/3,1/3)=\pi.
    \]
  \pause
  \item Starting from state 1 gives
    \[
      P_0=(1,0),\qquad P_1=(0.8,0.2),\qquad
      P_2=(0.72,0.28),\quad\ldots
    \]
    and $P_t$ approaches $\pi$
  \item<4-> Stationary means that applying the transition leaves the
    distribution unchanged; convergence means approaching it from another
    starting distribution
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Markov chain Monte Carlo (MCMC)}

  \begin{itemize}
  \item Produce draws $\theta^{(t)}$ given $\theta^{(t-1)}$ from a
    Markov chain, with {\color{uured}stationary distribution} $p(\theta|y)$
    \begin{itemize}
    \item<2->[+] generic
    \item<2->[+] combine sequence of easier Monte Carlo draws to form a Markov chain
    \item<3->[+] converges (in total variation) to $p(\theta|y)$ as $t\rightarrow\infty$
    \item<4->[+] for well-behaved chains, a central limit theorem holds
      for Monte Carlo averages
    \item<5->[-] draws are dependent
    \item<5->[-] constructing an \uured{efficient} Markov chain is not always easy
    \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}

\frametitle{MCMC summary}

  \begin{itemize}
  \item Random variables $\theta^{(1)},\theta^{(2)},\ldots$ where
    $\theta^{(t)}$ depends only on the previous state $\theta^{(t-1)}$
    \begin{equation*}
      p(\theta^{(t)}|\theta^{(1)},\ldots,\theta^{(t-1)})
      =p(\theta^{(t)}|\theta^{(t-1)})
    \end{equation*}
    \pause
  \item Chain has to be initialized with some starting point $\theta^{(0)}$ \pause
  % \item Transition distribution $T_t(\theta_t|\theta_{t-1})$ (may depend on $t$) \pause
  \item Choose a transition distribution so the
    {\color{uured}stationary distribution} of the Markov chain is $p(\theta|y)$
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Gibbs sampling}

  \begin{itemize}
  \item Alternate sampling from conditional distributions
  \item For two parameters, one complete sweep is
    \begin{align*}
      \theta_1^{(t)}&\sim
        p(\theta_1\mid\theta_2^{(t-1)},y),\\
      \theta_2^{(t)}&\sim
        p(\theta_2\mid\theta_1^{(t)},y).
    \end{align*}
  \pause
  \item More generally, update $j=1,\ldots,J$ in order using
    \[
      \theta_j^{(t)}\sim p\!\left(\theta_j\mid
      \theta_1^{(t)},\ldots,\theta_{j-1}^{(t)},
      \theta_{j+1}^{(t-1)},\ldots,\theta_J^{(t-1)},y\right).
    \]
  \pause
  \item Under standard conditions, the distribution of $\theta^{(t)}$
    converges to $p(\theta\mid y)$ as $t\rightarrow\infty$
  \item Several parameters can be updated together in a block
%  \item Related to the (stochastic) EM algorithm
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Example: Bivariate Normal using Gibbs}

\[
\theta \mid y \;\sim\; \mathcal{N}\!\left( \bar{y}, \tfrac{1}{n}\Sigma \right),
\qquad
\Sigma=
\begin{pmatrix}
  \sigma_1^2 & \rho\sigma_1\sigma_2\\
  \rho\sigma_1\sigma_2 & \sigma_2^2
\end{pmatrix}.
\]
Here $\sigma_1$ and $\sigma_2$ are marginal standard deviations,
$\rho$ is the correlation, and $\Sigma$ is known.
\vspace{3mm}
\pause

Using \uured{Gibbs sampling}:\\
\vspace{3mm}
Given starting values \((\theta_1^{(0)}, \theta_2^{(0)})\), for \(t = 1,2,\dots\):

\[
\theta_1^{(t)} \;\sim\; \mathcal{N}\!\left(
\bar{y}_1 + \rho\frac{\sigma_1}{\sigma_2}
  \big(\theta_2^{(t-1)} - \bar{y}_2\big),\;
\frac{\sigma_1^2}{n}(1-\rho^2)
\right),
\]

\[
\theta_2^{(t)} \;\sim\; \mathcal{N}\!\left(
\bar{y}_2 + \rho\frac{\sigma_2}{\sigma_1}
  \big(\theta_1^{(t)} - \bar{y}_1\big),\;
\frac{\sigma_2^2}{n}(1-\rho^2)
\right).
\]

\end{frame}

\subsection{Metropolis-Hastings}
\frame{\subsectionpage}

\begin{frame}

\frametitle{The Metropolis algorithm}

  \begin{itemize}
  \item Algorithm
    \begin{itemize}
      \item[1.] starting point $\theta^{(0)}$
      \item[2.] $t=1,2,\ldots$
        \begin{itemize}
        \item[(a)] pick a proposal $\theta^{*}$ from a {\color{uured} proposal distribution}
          $J(\theta^{*}|\theta^{(t-1)})$. \\
          (Proposal distribution has to be symmetric, i.e.\\
          $J(\theta_a|\theta_b)=J(\theta_b|\theta_a)$, for all
          $\theta_a,\theta_b$)
        \item<2->[(b)] calculate acceptance ratio
          \begin{equation*}
            r=\frac{p(\theta^{*}|y)}{p(\theta^{(t-1)}|y)}
          \end{equation*}
          \vspace{-6mm}
        \item<3->[(c)] set
          \begin{equation*}
            \theta^{(t)}=
            \begin{cases}
              \theta^{*} & \text{with probability $\min(r,1)$}\\
              \theta^{(t-1)} & \text{otherwise}
            \end{cases}
          \end{equation*}
          \uncover<4>{
          i.e., if $p(\theta^{*}|y)>p(\theta^{(t-1)}|y)$, always accept the proposal \\
          \hspace{0.4cm}and otherwise accept the proposal with probability $r$}
      \end{itemize}
      \vspace{-1.5\baselineskip}
    \item<5-> rejection of a proposal increments the time $t$ also by one\\
      i.e., the new state is the same as the previous state
      \item<6-> step c is executed by generating a random number from
        $\U(0,1)$
      \item<7-> $p(\theta^*|y)$ and $p(\theta^{(t-1)}|y)$ have the same
        normalizing constant, so the unnormalized density $q(\cdot|y)$
        can be used: {\color{uured}the constant cancels}
    \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Rejection sampling vs. Metropolis rejection}

  \begin{itemize}
  \item In rejection sampling, a rejected proposal is discarded
    \begin{itemize}
    \item only accepted proposals are included in the output
    \item the next proposal is drawn independently from $g$
    \end{itemize}
  \pause
  \item In the Metropolis algorithm, a rejected proposal is discarded,
    but the current state is repeated
    \[
      \theta^{(t)}=\theta^{(t-1)}.
    \]
    \begin{itemize}
    \item the repeated state is a valid draw from the Markov chain
    \item it must be retained when computing posterior summaries
    \end{itemize}
  \pause
  \item Keeping only accepted Metropolis proposals changes the distribution
    and generally gives incorrect estimates
  \end{itemize}

\end{frame}

\begin{frame}{Proposal scale: discussion}

  Consider a random-walk Metropolis proposal
  \[
    \theta^*\sim\mathcal{N}(\theta^{(t-1)},\sigma_p^2 I).
  \]

  \begin{itemize}
  \item Predict what happens when $\sigma_p$ is much too small.
  \item Predict what happens when $\sigma_p$ is much too large.
  \item For each case, what do you expect for the acceptance rate, trace
    plot, and autocorrelation?
  \end{itemize}

  \uncover<2->{
  {\color{uured}After discussion:} Small proposals are often accepted but
  move slowly. Large proposals move farther when accepted, but are often
  rejected. Both choices can give high autocorrelation and low ESS.}

\end{frame}

 \begin{frame}

\frametitle{Why the Metropolis algorithm works}

  \small
  \begin{itemize}
  \item Intuitively, the chain spends more time in high-density regions
    \begin{itemize}
    \item proposals to higher density are always accepted
    \item proposals to lower density are accepted only sometimes
    \end{itemize}
  \pause
  \item Detailed balance will show that $p(\theta\mid y)$ is a
    stationary distribution of the Metropolis chain
  \pause
  \item Convergence to that stationary distribution also requires a
    well-behaved chain
    \begin{itemize}
    \item the proposal must allow the chain to reach all relevant regions
      of the target distribution
    \item the chain must not be trapped in a deterministic cycle
    \end{itemize}
  \pause
  \item<4-> These ideas are formalized by \uured{irreducibility} and
    \uured{aperiodicity}
  \item<5-> They depend on both the target and the proposal; they do not
    follow merely because the target is a proper distribution
  \end{itemize}

\end{frame}


\begin{frame}

\frametitle{Detailed balance for the Metropolis algorithm}

  \begin{itemize}
  \item For a symmetric proposal $J(\theta'\mid\theta)$, the Metropolis
    acceptance probability is
    \[
      \alpha(\theta,\theta')=
      \min\!\left(1,\frac{p(\theta'\mid y)}{p(\theta\mid y)}\right).
    \]
  \pause
  \item For $\theta'\neq\theta$, the probability flow from $\theta$ to
    $\theta'$ is
    \begin{align*}
      &p(\theta\mid y)J(\theta'\mid\theta)\alpha(\theta,\theta')\\
      &\qquad=J(\theta'\mid\theta)
        \min\!\left(p(\theta\mid y),p(\theta'\mid y)\right).
    \end{align*}
  \pause
  \item Symmetry of $J$ makes this expression unchanged when
    $\theta$ and $\theta'$ are exchanged. Hence
    \[
      p(\theta\mid y)T(\theta'\mid\theta)
      =p(\theta'\mid y)T(\theta\mid\theta').
    \]
  \item<4-> The self-transition caused by rejection balances trivially.
    Thus detailed balance holds for the full transition kernel $T$, and
    $p(\theta\mid y)$ is stationary
  \end{itemize}

\frametitle{Metropolis-Hastings algorithm}

  \begin{itemize}
  \item The ideal proposal is the target distribution itself
    \begin{itemize}
    \item $J(\theta^{*}|\theta)\equiv p(\theta^{*}|y)$ gives independent
      draws with acceptance probability 1, but is not usually feasible
    \end{itemize}
  \item<2-> A good proposal resembles the target distribution
  \item<3-> Proposal scale controls movement through the target
    \begin{itemize}
    \item small scale: many accepted steps, but each step moves little
    \item large scale: longer proposed steps, but many are rejected
    \end{itemize}
  \item<4-> There is no universal optimal acceptance rate
    \begin{itemize}
    \item it depends on the target, dimension, proposal, and quantity
      of interest
    \item compare proposals using ESS per computation time or MCSE
    \end{itemize}
  \item<5-> \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=RandomWalkMH&target=standard}{Interactive demonstration}
\end{itemize}

\begin{frame}

\frametitle{Gibbs sampling as a special case}

  \begin{itemize}
  \item Update one component while holding $\theta_{-j}$ fixed, proposing
    \[
      \theta_j^*\sim p(\theta_j\mid\theta_{-j},y).
    \]
  \pause
  \item The Metropolis--Hastings acceptance ratio is
    \begin{align*}
      r
      &=\frac{p(\theta_j^*,\theta_{-j}\mid y)\,
        p(\theta_j\mid\theta_{-j},y)}
        {p(\theta_j,\theta_{-j}\mid y)\,
        p(\theta_j^*\mid\theta_{-j},y)}\\
      &=1,
    \end{align*}
    because each joint posterior factors into the same marginal for
    $\theta_{-j}$ times its full conditional
  \item<3-> Therefore every exact Gibbs proposal is accepted
  \end{itemize}

\end{frame}

  \begin{itemize}
  \item Early draws can be strongly influenced by the starting value
    \uncover<2->{
      \begin{itemize}
      \item the chain may not yet have reached the typical posterior region
        \\

      \end{itemize}
      }
      \vspace{-.5\baselineskip}
    \item<3-> Warm-up, also called {\color{uured} burn-in}, is the initial
      phase before retained sampling
    \begin{itemize}
      \item do not use warm-up draws in posterior summaries
      \item algorithms may adapt proposal or tuning parameters during warm-up
      \end{itemize}
    \item<4-> Discarding warm-up does not guarantee good sampling
      \begin{itemize}
      \item the retained chains can still mix slowly or miss posterior regions
      \item convergence and efficiency diagnostics are still required
      \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Classical $\widehat R$: interpretation}

  \begin{itemize}
  \item The classical diagnostic compares a total variance estimate
    $\widehat{\operatorname{var}}^{+}$ with the within-chain variance $W$
    \[
      \widehat R=\sqrt{\frac{\widehat{\operatorname{var}}^{+}}{W}}.
    \]
  \pause
  \item $\widehat R$ close to 1 means the between-chain and within-chain
    variation are similar
  \item $\widehat R>1$ indicates that the chains disagree more than expected
    from their within-chain variation
  \pause
  \item A value close to 1 is necessary but not sufficient: also inspect
    trace plots, ESS, and MCSE
  \end{itemize}

\end{frame}

\begin{frame}<0>[label=backup_rhat_wb]

\frametitle{ $\widehat{R}$}

  \begin{itemize}
  \item $M$ chains, each having $N$ draws
  \item<2-> Within chains variance $W$
    \begin{equation*}
      W=\frac{1}{M}\sum_{m=1}^M s^2_m ,\text{ where }
      s^2_m=\frac{1}{N-1}\sum_{n=1}^N (\theta_{nm}-\bar{\theta}_{.m})^2
    \end{equation*}
  \item<3-> Between chains variance $B$
    \begin{align*}
      B&=\frac{N}{M-1}\sum_{m=1}^M
      (\bar{\theta}_{.m}-\bar{\theta}_{..})^2,\\
      &\text{ where } \bar{\theta}_{.m}=\frac{1}{N}\sum_{n=1}^N \theta_{nm}, \,
      \bar{\theta}_{..}=\frac{1}{M}\sum_{m=1}^M\bar{\theta}_{.m}
    \end{align*}
    %\vspace{-6mm}
    \begin{itemize}
      \item<4-> $B/N$ is variance of the means of the chains
    \end{itemize}
    \vspace{2mm}
  \item<5-> Estimate total variance
    $\var(\theta|y)$ as a weighted mean of $W$ and $B$
    \begin{equation*}
      \widehat{\var}^{+}(\theta|y) = \frac{N-1}{N}W+\frac{1}{N}B
    \end{equation*}
  \end{itemize}

\end{frame}


\begin{frame}<0>[label=backup_rhat_vplus]

\frametitle{ $\widehat{R}$}

  \begin{itemize}
  \item Estimate total variance
    $\var(\theta|y)$ as a weighted mean of $W$ and $B$
    \begin{equation*}
      \widehat{\var}^{+}(\theta|y) = \frac{N-1}{N}W+\frac{1}{N}B
    \end{equation*}
    \vspace{-0.5\baselineskip}
    \begin{itemize}
    \item with overdispersed starting points and incomplete mixing, this
      typically \emph{overestimates} the marginal posterior variance
    \end{itemize}
    \vspace{0.5\baselineskip}
  \item<2-> Before chains have fully explored the posterior, $W$ typically
    \emph{underestimates} the marginal posterior variance
    \begin{itemize}
    \item single chains have not yet visited all points in the distribution
    \item when $N\rightarrow\infty, \quad \E(W)\rightarrow \var(\theta|y)$
    \end{itemize}
    \vspace{0.5\baselineskip}
  \item<3-> Comparing these two estimates motivates
    \begin{equation*}
      \widehat{R}=\sqrt{\frac{\widehat{\var}^{+}}{W}}
    \end{equation*}
\end{itemize}

\end{frame}

\subsection{$S_\text{eff}$, MCSE, and autocorrelation}
\frame{\subsectionpage}


\begin{frame}

\frametitle{MCMC draws are dependent}

  \begin{itemize}
    \item Notation: $M$ chains, $N$ retained draws per chain, and
      $S=MN$ retained draws in total
    \item For well-behaved chains, Monte Carlo averages converge to
      posterior expectations
      \begin{align*}
        E_{\color{blue} p(\theta|y)}[f(\theta)] \approx \frac{1}{S} \sum_{s=1}^S f(\theta^{(s)})
      \end{align*}
    \item A Markov chain central limit theorem can then be used to estimate
      Monte Carlo error, but dependence makes this more difficult
      \begin{itemize}
        \item evaluation of {\it effective} sample size, $S_\text{eff}$.
      \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Autocorrelation in MCMC}

  \begin{itemize}
  \item The autocorrelation function measures dependence between draws
    separated by a given lag
    \begin{itemize}
    \item high autocorrelation means that successive draws contain less
      new information
    \item it helps compare the efficiency of MCMC methods
    \end{itemize}
  \end{itemize}

\begin{frame}

\frametitle{Effective sample size and MCSE}

  \begin{itemize}
  \item Suppose we estimate $\mu_f=\E[f(\theta)\mid y]$ using
    \[
      \widehat\mu_f=\frac{1}{S}\sum_{s=1}^S f(\theta^{(s)}).
    \]
  \pause
  \item If the autocorrelation time for $f(\theta)$ is $\tau_f$, define
    \[
      S_{\mathrm{eff},f}=\frac{S}{\tau_f}.
    \]
  \pause
  \item The Monte Carlo standard error is approximately
    \[
      \operatorname{MCSE}(\widehat\mu_f)
      \approx \frac{\operatorname{sd}(f(\theta))}
      {\sqrt{S_{\mathrm{eff},f}}}.
    \]
  \item<4-> Larger effective sample size means a more precise Monte Carlo
    estimate; doubling ESS reduces MCSE by a factor of $\sqrt{2}$
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Autocorrelation}

  \begin{itemize}
  \item For lag $k$, estimate the expected squared change within chains
    \begin{equation*}
      \widehat V_k=\frac{1}{M(N-k)}
      \sum_{m=1}^M\sum_{n=1}^{N-k}
      \left(\theta_{n+k,m}-\theta_{n,m}\right)^2.
    \end{equation*}
  \pause
  \item Estimate the lag-$k$ autocorrelation by
    \begin{equation*}
      \widehat\rho_k
      =1-\frac{\widehat V_k}{2\widehat{\operatorname{var}}^{+}}.
    \end{equation*}
  \item<3-> Small squared changes imply high autocorrelation; large changes
    imply lower autocorrelation
  \item<4-> Using $\widehat{\operatorname{var}}^{+}$ incorporates information
    from all chains and penalizes chains that have not mixed well
 \end{itemize}
\end{frame}



\begin{frame}

\frametitle{Integrated autocorrelation time}

  \begin{itemize}
  \item Let $\rho_k$ be the autocorrelation at lag $k$. The integrated
    autocorrelation time is
    \begin{align*}
      \tau = 1 + 2 \sum_{k=1}^\infty \rho_k,
      \qquad S_\text{eff}=\frac{S}{\tau}.
    \end{align*}
  \pause
  \item $\tau$ measures how many correlated draws provide roughly the
    same information as one independent draw
  \pause
  \item Estimated autocorrelations become noisy at long lags
    \begin{itemize}
    \item practical estimators truncate the sum at a finite lag $K$
    \begin{align*}
      \hat{\tau} = 1 + 2 \sum_{k=1}^K \hat{\rho}_k
    \end{align*}
    \item statistical software chooses the maximum lag $K$ adaptively
    \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}<0>[label=backup_geyer]

\frametitle{Geyer's adaptive window estimator of $\tau$}

  \begin{itemize}
  \item The maximum lag $K$ can be chosen adaptively
    \begin{itemize}
    \item for a stationary, reversible Markov chain
    \item let $\Gamma_m=\rho_{2m}+\rho_{2m+1}$, the sum of two
      consecutive autocorrelations
    \item $\Gamma_m$ is a positive, decreasing, and convex function of $m$
    \end{itemize}
    \vspace{0.5\baselineskip}
  \item<2-> Initial positive sequence estimator (Geyer's IPSE)
      \begin{itemize}
        \item Choose the largest $m$ such that all values of the sequence
        $\hat{\Gamma}_1, \ldots, \hat{\Gamma}_m$ are positive
      \end{itemize}
  \vspace{0.5\baselineskip}
      \includegraphics[width=6cm]{figs/Metrop1acf.pdf}
  \end{itemize}
\end{frame}

\section{Difficult geometries}
\frame{\sectionpage}

\begin{frame}

\frametitle{Difficult geometry: changing scale and direction}

  \begin{itemize}
  \item \uured{Nonlinear dependence}: posterior mass follows a curved,
    narrow region
    \begin{itemize}
    \item a fixed random-walk direction repeatedly proposes points outside
      the high-density region
    \item symptoms: slow movement, high autocorrelation, and low ESS
    \end{itemize}
  \pause
  \item \uured{Funnel geometry}: the appropriate step size changes across
    the posterior
    \begin{itemize}
    \item a step size suitable for the wide part is too large for the neck
    \item a step size suitable for the neck moves very slowly in the wide part
    \item symptoms: chains explore different scales or rarely enter the neck
    \end{itemize}
  \end{itemize}

  \vfill
  \centering
  \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=RandomWalkMH&target=banana}{Banana demo}
  \qquad
  \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=RandomWalkMH&target=funnel}{Funnel demo}

\end{frame}

\begin{frame}

\frametitle{Difficult geometry: hidden regions}

  \begin{itemize}
  \item \uured{Multimodality}: chains can remain in different modes
    \begin{itemize}
    \item chain means disagree and $\widehat R$ can remain above 1
    \item if every chain starts in the same mode, diagnostics may fail to
      reveal the missing mode
    \end{itemize}
  \pause
  \item \uured{Non-identifiability}: many parameter combinations fit the
    data almost equally well
    \begin{itemize}
    \item the posterior contains a ridge rather than one compact region
    \item symptoms: drifting traces, strong dependence, and low ESS
    \end{itemize}
  \pause
  \item \uured{Heavy tails}: occasional extreme draws dominate summaries
    \begin{itemize}
    \item means or variances may not exist, so the usual CLT and MCSE may
      not apply
    \end{itemize}
  \end{itemize}

  \vfill
  \centering
  \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=RandomWalkMH&target=multimodal}{Multimodal demo}
  \qquad
  \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=RandomWalkMH&target=donut}{Ridge demo}

\end{frame}

\begin{frame}

\frametitle{MCMC diagnostic workflow}

  \begin{enumerate}
  \item Run several chains from different, reasonably dispersed starting
    values
  \item Remove warm-up draws before computing posterior summaries
  \item Inspect trace plots: chains should explore the same regions without
    persistent trends
  \item Check that $\widehat R$ is close to 1 for all reported quantities
  \item Check ESS and MCSE for the posterior summaries that matter
  \item If diagnostics are poor, investigate the proposal, parameterization,
    model, and posterior geometry
  \end{enumerate}

  \pause
  \vspace{0.5\baselineskip}
  {\color{uured}Running longer helps only if the chain is exploring the
  relevant posterior regions.}

\end{frame}
