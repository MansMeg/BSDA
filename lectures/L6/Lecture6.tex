\begin{frame}
\frametitle{Recap: Markov chain Monte Carlo}

\begin{itemize}
\item The \uured{transition distribution} $T$ describes the next state
  given the current state.
\item Construct $T$ so that $p(\theta\mid y)$ is its
  \uured{stationary distribution}.
\pause
\item Under suitable conditions, the chain approaches this distribution
  from its starting point.
\item[+] A general approach when direct posterior sampling is difficult.
\pause
\item[-] Draws are dependent: precision depends on the
  effective sample size (ESS), $S_{\mathrm{eff}}$.
\item[-] Assess convergence and the Monte Carlo standard error (MCSE)
  of each estimated summary.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Recap: Gibbs and Metropolis}

\begin{itemize}
\item Gibbs sampling
\begin{itemize}
\item Update one parameter (or a block) from its full conditional:
\[
\theta_j \sim p(\theta_j|\theta_{-j}, y)
\]
\item[+] Often easy to construct when conditionals are available.
\item[-] Single-parameter updates can mix slowly under strong correlation.
\end{itemize}
\pause
\item Metropolis(-Hastings) sampling
\begin{itemize}
\item Propose a joint (or block) move from $J$.
\item The proposal \uured{plus acceptance or rejection} defines $T$;
  on rejection, the next state equals the current state.
\pause
\item[+] A well-chosen joint proposal can handle correlation efficiently.
\item[-] Efficiency depends on proposal shape and scale, especially in
  high dimensions.
\end{itemize}
\end{itemize}

\end{frame}

\section{Hamiltonian Monte Carlo}
\frame{\sectionpage}

% Motivation: Why HMC?
\begin{frame}
\frametitle{Why Hamiltonian Monte Carlo?}

\begin{itemize}
\item Want to build an \uured{efficient} Markov chain
\begin{itemize}
\item We want to sample all components of $\theta$ jointly.
\item We know the \uured{unnormalized} posterior
  $q(\theta|y)=Z \cdot p(\theta|y)$, where $Z$ does not depend on $\theta$.
\item Standard Hamiltonian Monte Carlo (HMC) needs continuous unknowns and a sufficiently smooth
  log density.
\item Can the \uured{gradient} of $\log q(\theta|y)$ guide efficient proposals?
\pause
\item \uured{Hamiltonian Monte Carlo}!
\end{itemize}
\end{itemize}

\end{frame}


\begin{frame}
\frametitle{What is Hamiltonian Monte Carlo?}

\begin{itemize}
\item Add momentum variables to our posterior (\uured{canonical} distribution)
\[
 p(\theta,\psi | y) = p(\theta | y) \cdot p(\psi | \theta , y)\,,
\]
in practice we let $p(\psi | \theta , y) = p(\psi)$
\pause
\item Idea from physics (mechanics):
\begin{itemize}
\item $\theta$: position
\item $\psi$: momentum
\end{itemize}
\pause
\item Define the Hamiltonian as
\begin{align}
H(\theta, \psi)  & = - \log(p(\psi)) - \log(p(\theta | y))\\
 & = K(\psi) + V(\theta)\,,
\end{align}
where $K(\psi)$ is the \uured{kinetic} energy and $V(\theta)$ is the \uured{potential} energy
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Why can we add momentum?}

\begin{itemize}
\item Choose any proper momentum density $p(\psi)$, independent of $\theta$,
  and define the joint target
  \[
  p(\theta,\psi\mid y)=p(\theta\mid y)p(\psi).
  \]
\pause
\item Its marginal distribution for $\theta$ is unchanged:
  \begin{align*}
  \int p(\theta,\psi\mid y)\,d\psi
    &=p(\theta\mid y)\int p(\psi)\,d\psi\\
    &=p(\theta\mid y).
  \end{align*}
\item Hence we can sample the joint target and \uured{keep only $\theta$}.
\pause
\item Refreshing $\psi\sim p(\psi)$ is a draw from its full conditional.
  It changes the trajectory without changing the target posterior.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{What is Hamiltonian Monte Carlo?}

\begin{itemize}
\item Hamiltonian dynamics (\uured{preserve energy})
\begin{align}
\frac{d\theta}{dt} &=  \frac{\partial H}{\partial \psi} = \frac{\partial K}{\partial \psi}\\
\frac{d\psi}{dt} &=  - \frac{\partial H}{\partial \theta} = - \frac{\partial V}{\partial \theta}
\end{align}
\item For computing the dynamics, additive constants do not matter:
\[
V(\theta) = -\log q(\theta\mid y)
          = -\log p(\theta)-\log p(y\mid\theta).
\]
\item Let $\psi\sim N(0,M)$, with positive-definite \uured{mass matrix} $M$.
\[
K(\psi)=-\log p(\psi)=\tfrac12\psi^\top M^{-1}\psi+C,
\]
where $C$ does not depend on $\psi$.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Normal example: constant energy}

For $\theta\sim N(0,1)$ and $\psi\sim N(0,1)$ independently,
\[
H(\theta,\psi)=\frac{\theta^2+\psi^2}{2}
\quad\text{(ignoring an additive constant).}
\]
\begin{columns}[c,onlytextwidth]
\begin{column}{0.48\textwidth}
\includegraphics[width=\linewidth]{figs/hmc_normal_energy.pdf}
\end{column}
\begin{column}{0.49\textwidth}
\[
\frac{d\theta}{dt}=\psi,\qquad
\frac{d\psi}{dt}=-\theta.
\]
\begin{itemize}
\item Exact dynamics follow a circle, clockwise: energy is constant.
\item The red arrow refreshes momentum at fixed $\theta$.
\item The next trajectory can follow a different energy circle.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{Choosing the mass matrix}

\begin{itemize}
\item The mass matrix controls how momentum moves the parameters:
\[
\frac{d\theta}{dt}=M^{-1}\psi.
\]
\item Write $\Sigma=\operatorname{Cov}(\theta\mid y)$.
  For a roughly Gaussian posterior, a useful choice is
\[
M^{-1}\approx\Sigma.
\]
\pause
\item During warmup, Stan estimates a regularized approximation:
\begin{itemize}
\item diagonal: accounts for different parameter scales;
\item dense: also accounts for linear correlations.
\end{itemize}
\item In Stan these calculations use \uured{unconstrained coordinates}.
  One fixed matrix cannot remove all nonlinear geometry.
\end{itemize}
\end{frame}


\begin{frame}

\frametitle{The leapfrog integrator}

\begin{itemize}
\item A \uured{discrete} approximation to Hamiltonian dynamics:
\[
\frac{d\theta}{dt}=M^{-1}\psi,
\qquad
\frac{d\psi}{dt}=\nabla_\theta\log q(\theta\mid y).
\]
\pause
\item For $\ell=0,\ldots,L-1$, repeat \uured{all three updates}:
\begin{align*}
\psi_{\ell+1/2} &= \psi_\ell
  +\frac{\epsilon}{2}\nabla_\theta\log q(\theta_\ell\mid y),\\
\theta_{\ell+1} &= \theta_\ell+\epsilon M^{-1}\psi_{\ell+1/2},\\
\psi_{\ell+1} &= \psi_{\ell+1/2}
  +\frac{\epsilon}{2}\nabla_\theta\log q(\theta_{\ell+1}\mid y).
\end{align*}
\item The final gradient uses the \uured{updated position} $\theta_{\ell+1}$.
\pause
\item Step size $\epsilon$ controls numerical accuracy;
  $L\epsilon$ is the total integration time.
\item Longer trajectories are not automatically harmless: errors can
  grow, especially when the step size is too large.
\end{itemize}
\end{frame}

\begin{frame}

\frametitle{HMC: construct a proposal}

At iteration $t$, start at $\theta_{t-1}$ with positive target density.
\begin{enumerate}
\item Refresh momentum:
\[
\psi_0 \sim N(0,M).
\]
\pause
\item Starting at $(\theta_{t-1},\psi_0)$, take $L$ leapfrog steps
  of size $\epsilon$ to get $(\theta^\star,\widetilde\psi)$.
\pause
\item Reverse the final momentum: $\psi^\star=-\widetilde\psi$.
  This makes the proposal reversible.
\end{enumerate}
\begin{itemize}
\item For Gaussian momentum, $K(-\psi)=K(\psi)$, so the reversal
  does not change the energy or acceptance probability.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{HMC: accept or retain the old state}

\begin{itemize}
\item Calculate the energy change:
\[
\Delta H=H(\theta^\star,\psi^\star)
         -H(\theta_{t-1},\psi_0).
\]
\item The Metropolis acceptance probability is
\[
r=\min\!\left(1,
\frac{q(\theta^\star\mid y)p(\psi^\star)}
     {q(\theta_{t-1}\mid y)p(\psi_0)}\right)
 =\min(1,e^{-\Delta H}).
\]
\pause
\item Draw $U\sim\operatorname{Uniform}(0,1)$ and compare on the log scale:
\[
\theta_t=\begin{cases}
\theta^\star, & \log U<\min(0,-\Delta H),\\
\theta_{t-1}, & \text{otherwise}.
\end{cases}
\]
\item As in L4, the log comparison avoids exponentiating extreme values.
  Discard the momentum; refresh it next iteration.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Numerical HMC: one leapfrog step}

\begin{itemize}
\item Target $\theta\sim N(0,1)$ and mass $M=1$, so
  $\nabla_\theta\log q(\theta)=-\theta$.
\item Suppose $\theta_{t-1}=1$ and refreshed momentum $\psi_0=1$.
  Take one leapfrog step: $\epsilon=1$, $L=1$.
\end{itemize}
\pause
The momentum half-step, position full-step, and momentum half-step give
\begin{align*}
\psi_{1/2}
  &=\psi_0-\frac{\epsilon}{2}\theta_{t-1}
    =1-\frac12\cdot1=0.5,\\[0.5em]
\theta^\star
  &=\theta_{t-1}+\epsilon\psi_{1/2}
    =1+1\cdot0.5=1.5,\\[0.5em]
\widetilde\psi
  &=\psi_{1/2}-\frac{\epsilon}{2}\theta^\star
    =0.5-\frac12\cdot1.5=-0.25.
\end{align*}
\pause
\begin{itemize}
\item Reverse momentum: $\psi^\star=-\widetilde\psi=0.25$.
  The joint proposal is $(\theta^\star,\psi^\star)=(1.5,0.25)$.
\item Here $\epsilon$ and $L$ are chosen for simple arithmetic.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Numerical HMC: accept or reject}

Ignoring an additive constant, $H(\theta,\psi)=(\theta^2+\psi^2)/2$.
\begin{align*}
H(\theta_{t-1},\psi_0)&=\frac{1^2+1^2}{2}=1,\\
H(\theta^\star,\psi^\star)&=\frac{1.5^2+0.25^2}{2}=1.15625.
\end{align*}
\pause
\begin{itemize}
\item Leapfrog has increased the energy slightly:
\[
\Delta H=1.15625-1=0.15625.
\]
\item The acceptance probability is
\[
r=\min(1,e^{-0.15625})\approx0.8553.
\]
\pause
\item Draw $U\sim\operatorname{Uniform}(0,1)$. If $U=0.80$,
  then $U<r$: \uured{accept} and set $\theta_t=1.5$.
\item If instead $U=0.90$, \uured{reject} and retain $\theta_t=1$.
  In either case, refresh momentum at the next iteration.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Pair discussion: step size and exploration}

Discuss in pairs (1--2 minutes).
\begin{itemize}
\item Consider HMC with $\epsilon=0.1$ and $L=20$.
  We halve $\epsilon$ but keep $L$ fixed.
\item What happens to numerical accuracy and integration time?
  Does this automatically improve posterior exploration?
\item How could we preserve the original integration time?
\end{itemize}

\uncover<2->{
\vspace{0.5\baselineskip}
\uured{After discussion:}
\begin{itemize}
\item Smaller steps generally improve numerical accuracy,
  but $L\epsilon$ falls from $2$ to $1$.
\item A shorter trajectory may limit movement, so better accuracy
  does not guarantee better exploration.
\item Double $L$ to $40$ to keep $L\epsilon=2$, at the cost of
  more gradient evaluations per proposal.
\end{itemize}}
\end{frame}

\begin{frame}
\frametitle{Why does the HMC update work?}

\begin{itemize}
\item \uured{Reversibility}: leapfrog followed by a momentum reversal can
  be undone by applying the same proposal again.
\pause
\item \uured{Volume preservation}: leapfrog does not expand or compress
  volumes in $(\theta,\psi)$ space. No proposal Jacobian is needed.
\pause
\item \uured{Metropolis correction}: the acceptance step corrects the
  numerical energy error, preserving the joint target
  \[
  p(\theta,\psi\mid y)\propto e^{-H(\theta,\psi)}.
  \]
\item Marginalizing momentum then gives $p(\theta\mid y)$.
  Energy conservation alone is not the full justification.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Dynamic HMC}

  \begin{itemize}
  \item \uured{Goal:} Simplify/adapt the tuning of HMC
  \item \uured{Dynamic}: choose the number of leapfrog steps $L$
    within each iteration.
  \item The \uured{No-U-Turn Sampler (NUTS)} grows a trajectory in randomly
    chosen forward or backward directions, doubling its length in stages.
  \pause
  \item \uured{U-turn intuition}, with $M=I$ and one moving endpoint:
  \[
  (\theta_L-\theta_{\mathrm{start}})^\top\psi_L<0.
  \]
  The motion now points back towards the starting position.
  \item This is an illustration, \uured{not the full stopping rule}:
    NUTS checks both ends and parts of the growing trajectory.
  \pause
  \item Select the next state from eligible trajectory states using a
    rule designed to preserve the target distribution.
    These are not separately accepted MCMC draws.
  \item Step size still controls integration accuracy and computational cost.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Dynamic Hamiltonian Monte Carlo summary}

\begin{itemize}
\item Parameters:
\begin{itemize}
\item[$\epsilon$] step size \pause
\item[$M$] mass matrix\pause
\end{itemize}
\item[+] Can be very efficient ($S_{\mathrm{eff}}$)
\item[+] Additional diagnostics
\pause
\item $L$ is chosen anew at each iteration, including after warmup.
\item Stan transforms bounded parameters to unconstrained coordinates.
\item Still requires continuous unknowns and a smooth log density.
  Discrete data are fine; discrete latent parameters need another strategy.
\item[-] Each iteration still requires many gradient evaluations.
\pause
\end{itemize}

\begin{frame}

\frametitle{Maximum tree depth}

  \begin{itemize}
  \item NUTS reached its computational limit before its usual stopping rule.
  \item Primarily an \uured{efficiency warning}: repeated truncation can
    increase autocorrelation and reduce ESS.
  \item This alone does not demonstrate biased sampling or failure to
    explore the posterior.
  \pause
  \item Check model specification, scaling and parameterization;
    examine the other diagnostics too.
  \item Increasing \texttt{max\_treedepth} permits more computation,
    but does not repair poor posterior geometry.
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Divergent transitions}

% What
\begin{itemize}
  \item HMC-specific diagnostic
  \pause
  \item The Hamiltonian $H(\theta, \psi) $ should remain roughly constant throughout the trajectory:
  \[
  H(\theta, \psi) \approx H(\theta_{\mathrm{start}}, \psi_{\mathrm{start}})
  \]
  \item Stan flags an excessively large positive increase at any point along the trajectory:
  \[
  \Delta H = H(\theta, \psi) - H(\theta_{\mathrm{start}}, \psi_{\mathrm{start}}) \gg 0
  \]
  \item The numerical trajectory is unreliable, often because the
    posterior curvature changes rapidly.
  \item Divergences after warmup warn of possible biased estimates:
    some regions may be poorly explored.
\end{itemize}
%\end{frame}

% Why


\end{frame}

\begin{frame}
\frametitle{Divergent transitions}

% Why
\begin{itemize}
  \item A step size that works in one region may be too large in another.
  \item In a highly curved region, small changes in position can produce
    large changes in the \uured{gradient}.
  \pause
  \item The gradient determines the \uured{change in momentum};
    it is not the momentum itself:
    \[
    \frac{d\psi}{dt}=\nabla_\theta\log q(\theta\mid y).
    \]
  \item Narrow regions and changing scales, such as funnels in
    hierarchical models, can make leapfrog unstable.
  \item Strong linear correlation can often be handled by the mass
    matrix; changing local curvature is harder.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Divergent transitions}

% Solutions
\begin{itemize}
  \item When divergences remain after warmup:
  \begin{enumerate}
    \item Check the model and code, parameter scales, and whether the
      priors provide sensible regularization.
    \pause
    \item Reparameterize to make the geometry easier to traverse,
      for example with a non-centered hierarchical model.
    \pause
    \item Consider a smaller step size. In Stan, a higher
      \texttt{adapt\_delta} usually leads to smaller steps during warmup.
    \item Refit and check whether divergences disappear without making
      sampling impractically slow.
    \end{enumerate}
  \item More leapfrog steps \uured{at the same step size} do not fix
    inaccurate integration. Longer chains do not remove this bias.
\end{itemize}

\end{frame}

\frametitle{Difficult posterior geometry}

  \begin{itemize}
  \item \uured{Nonlinear dependence}: the direction of the narrow
    posterior region changes with position.
    A fixed mass matrix cannot straighten every part of it.
  \pause
  \item \uured{Funnels}: a step size suitable for the wide region may
    be too large in the narrow region.
  \pause
  \item \uured{Separated modes}: HMC may also struggle to cross
    low-density regions between modes.
  \end{itemize}

\begin{frame}
\frametitle{Identifiability and heavy tails}

  \begin{itemize}
  \item \uured{Non-identifiability}: distinct parameter values imply the
    same data distribution; weak identification gives similar likelihoods.
    \begin{itemize}
    \item With weak priors, this can produce posterior ridges or separated modes.
    \item Priors can constrain these directions; a connected set of
      posterior modes is not guaranteed.
    \end{itemize}
  \pause
  \item \uured{Heavy tails}: the difficulty depends on the summary
    $f(\theta)$ being estimated.
    \begin{itemize}
    \item If its posterior mean does not exist, there is no finite
      expectation to estimate.
    \item Infinite variance rules out the usual finite-variance central
      limit theorem (CLT) and its MCSE formula.
      Other summaries may still be well behaved.
    \end{itemize}
  \end{itemize}

  \vfill
  \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=RandomWalkMH&target=donut}{Ridge demo: random-walk Metropolis, as in L5}
\end{frame}

\begin{frame}
\frametitle{Diagnostic recap: can we trust the draws?}

\begin{itemize}
\item Run multiple chains, preferably at least four, with dispersed
  initial values. Discard warmup.
\item Inspect trace plots and require $\widehat R<1.01$ for the
  parameters and derived quantities of interest.
\pause
\item Check ESS and MCSE \uured{for each reported summary}.
  Using the classical ESS from L5, when the usual CLT applies,
  \[
  \operatorname{MCSE}(\widehat\mu_f)
  \approx \frac{\operatorname{SD}(f(\theta)\mid y)}{\sqrt{S_{\mathrm{eff},f}}}.
  \]
\item Investigate post-warmup divergences before trusting estimates.
  Tree-depth warnings primarily concern efficiency.
\pause
\item Good $\widehat R$ and ESS do not cancel a divergence warning.
  No diagnostic proves that all important regions were explored.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Pair discussion: conflicting diagnostics}

Discuss in pairs (1--2 minutes).
\begin{itemize}
\item Four chains give $\widehat R=1.00$ and ESS $=2000$ for a
  parameter of interest, but report 12 post-warmup divergences.
\item Can we trust its posterior summaries?
\item Would simply running the chains longer solve the problem?
\end{itemize}

\uncover<2->{
\vspace{0.5\baselineskip}
\uured{After discussion:}
\begin{itemize}
\item Good $\widehat R$ and ESS do not rule out biased estimates
  when divergences remain.
\item Longer chains do not repair inaccurate numerical integration
  or the underlying difficult geometry.
\item Check the model, scales and priors. Consider reparameterization
  or smaller steps, then refit and check again.
\end{itemize}}
\end{frame}

\begin{frame}
\frametitle{From model to checked predictions}

\begin{enumerate}
\item \uured{Specify}: observation model, parameter constraints and
  scientifically meaningful priors.
\item \uured{Fit}: run several chains with warmup; use a known-answer
  test when one is available.
\item \uured{Check computation}: traces, $\widehat R$, ESS, MCSE,

\item \uured{Interpret and predict}: summarize parameters and generated

\item \uured{Check the model}: compare predictive implications with
  observed data. Accurate computation does not guarantee a good model.
\end{enumerate}

\begin{frame}
\frametitle{Adaptive dynamic HMC in Stan}

  \begin{itemize}
  \item \uured{During warmup}, Stan adapts the mass matrix and step size.
    \begin{itemize}
    \item The mass matrix accounts for parameter scales and, if dense,
      linear correlations.
    \item \texttt{adapt\_delta} is the target average acceptance statistic
      used to tune the step size.
    \item A higher target usually gives smaller steps and more computation.
    \end{itemize}
  \pause
  \item \texttt{adapt\_delta} is \uured{not a guarantee} that this fraction
    of final NUTS draws will be accepted.
  \pause
  \item \uured{After warmup}, the adapted mass matrix and step size
    stop adapting.
    \begin{itemize}
    \item NUTS still chooses a trajectory length $L$ at \uured{each iteration}.
    \item \texttt{max\_treedepth} limits the work per iteration.
    \end{itemize}
  \item Use the post-warmup draws for inference.
\end{itemize}

parameters {

\end{lstlisting}}

  \begin{itemize}
  \item Stan samples an unconstrained variable $z$, then maps it back
    to the declared parameter $\theta$.
  \item For the interiors of the declared bounds:
    \[
    \begin{array}{ll}
    \texttt{<lower=a>}: & z=\log(\theta-a),\\[6pt]
    \texttt{<lower=a,upper=b>}: &
    z=\operatorname{logit}\!\left(\dfrac{\theta-a}{b-a}\right).
    \end{array}
    \]
  \item Here $a=0$, $b=1$, so $z=\operatorname{logit}(\theta)$.
  \item<2-> Stan automatically adds the log-Jacobian adjustment for
    these \uured{declared parameter constraints} (BDA3, p.\ 21).
  \end{itemize}

\begin{frame}
\frametitle{Binomial example: an exact benchmark}

\begin{itemize}
\item In our example, $N=10$, $y=7$, and $\theta\sim\operatorname{Beta}(1,1)$.
\item Bayes' rule gives
  \begin{align*}
  p(\theta\mid y)&\propto
    \theta^7(1-\theta)^3\cdot\theta^0(1-\theta)^0,\\
  \theta\mid y&\sim\operatorname{Beta}(8,4).
  \end{align*}
\pause
\item We know the posterior mean and standard deviation (SD) exactly:
  \[
  E(\theta\mid y)=\frac{8}{12}=\frac23,
  \quad
  \operatorname{SD}(\theta\mid y)
    =\sqrt{\frac{8\cdot4}{12^2\cdot13}}\approx0.1307.
  \]
\item Stan's estimates should agree \uured{within Monte Carlo error},
  not match every digit.
\end{itemize}
\end{frame}

\begin{frame}[fragile]

\frametitle{Gaussian regression: the observation model}

\begin{itemize}
\item At known predictor values $x_i$, assume conditional independence:
\[
y_i\mid\alpha,\beta,\sigma,x_i
  \sim N(\alpha+\beta x_i,\sigma^2),\qquad i=1,\ldots,N.
\]
\item $\alpha$ is the intercept, $\beta$ the slope, and $\sigma$ the
  residual \uured{standard deviation}.
\pause
\item Set $\mu_i=\alpha+\beta x_i$. Stan's vectorized statement
\end{itemize}
{\small
\begin{lstlisting}[language=Stan]
y ~ normal(mu, sigma);
\end{lstlisting}
}
adds the sum of the individual log densities:
\[
\log p(y\mid\alpha,\beta,\sigma,x)
  =\sum_{i=1}^N\log N(y_i\mid\mu_i,\sigma^2).
\]
Stan takes $\sigma$, not $\sigma^2$, as the second argument.
\end{frame}

\begin{frame}[fragile]
\frametitle{Gaussian regression: posterior prediction}

Add \texttt{real xpred;} to the data block, then append:
{\small
\begin{lstlisting}[language=Stan]
generated quantities {
  real mu_pred = alpha + beta*xpred;
  real y_pred = normal_rng(mu_pred, sigma);
}
\end{lstlisting}
}
\begin{itemize}
\item At each posterior draw, \texttt{mu\_pred} is the regression mean
  at the fixed prediction location.
\item \texttt{y\_pred} adds a new residual to simulate a
  \uured{future observation} at that location.
\item Keep both: they answer different questions.
  These lines use the Gaussian observation model.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Mean uncertainty versus predictive uncertainty}

At fixed $x_{\mathrm{pred}}$, write
$\mu_{\mathrm{pred}}=\alpha+\beta x_{\mathrm{pred}}$.
\begin{itemize}
\item The posterior of $\mu_{\mathrm{pred}}$ describes uncertainty
  about the \uured{regression mean}.
\item A new observation also varies around that mean:
\[
y_{\mathrm{pred}}\mid\alpha,\beta,\sigma
  \sim N(\mu_{\mathrm{pred}},\sigma^2).
\]
\pause
\item By the law of total variance, when these moments are finite,
\[
\underbrace{\operatorname{Var}(y_{\mathrm{pred}}\mid y)}_{\text{predictive uncertainty}}
=\underbrace{\operatorname{Var}(\mu_{\mathrm{pred}}\mid y)}_{\text{mean uncertainty}}
 +\underbrace{E(\sigma^2\mid y)}_{\text{residual variation}}.
\]
\item A predictive interval for a new observation accounts for both
  sources, not just uncertainty in the fitted line.
\end{itemize}
Predictor values are held fixed throughout.
\end{frame}

\begin{frame}[fragile]

\frametitle{Student-t linear model}
  {\footnotesize
  \begin{lstlisting}[language=Stan]
// Data and prior inputs as before
parameters {
  real alpha;
  real beta;
  real<lower=0> sigma;
  real<lower=1,upper=80> nu;
}
transformed parameters {
  vector[N] mu;
  mu = alpha + beta*x;
}
model {
  alpha ~ normal(pmualpha, psalpha);
  beta ~ normal(pmubeta, psbeta);
  sigma ~ normal(0, pssigma);
  nu ~ gamma(2, 0.1);
  y ~ student_t(nu, mu, sigma);
}
  \end{lstlisting}
}
\end{frame}

\begin{frame}
\frametitle{Student-t scale and degrees of freedom}
\begin{itemize}
\item $\mu$ is the location and $\sigma$ is a scale parameter, not generally the standard deviation.
\item Smaller $\nu$ gives heavier tails; large $\nu$ approaches the Gaussian model.
\item Here we restrict $1<\nu<80$:
  \begin{itemize}
  \item the lower bound ensures a finite conditional mean;
  \item the upper bound caps the range at nearly Gaussian errors.
  \end{itemize}
\item These bounds are modelling choices, not requirements of Stan.
\item \texttt{gamma(2, 0.1)} uses shape 2 and rate 0.1. Together with the bounds, this defines a truncated gamma prior.
\end{itemize}
\end{frame}

\frametitle{Standardized regression: data}

  {\footnotesize
\begin{lstlisting}[language=Stan]
data {
  int<lower=2> N;
  vector[N] x;
  vector[N] y;
  real xpred;
}
transformed data {
  real mx = mean(x);
  real my = mean(y);
  real sx = sd(x);
  real sy = sd(y);
  vector[N] x_std;
  vector[N] y_std;
  if (sx <= 0 || sy <= 0)
    reject("x and y must have positive SD");
  x_std = (x - mx) / sx;
  y_std = (y - my) / sy;
}
\end{lstlisting}
  }
\begin{itemize}
\item Center and scale once, using the training data.
  Model and prediction blocks follow on the next two slides.
\end{itemize}
\end{frame}

\begin{frame}[fragile]
\frametitle{Standardized regression: model}

{\small
\begin{lstlisting}[language=Stan]
parameters {
  real alpha_std;
  real beta_std;
  real<lower=0> sigma_std;
}
model {
  alpha_std ~ normal(0, 1);
  beta_std ~ normal(0, 1);
  sigma_std ~ normal(0, 1);
  y_std ~ normal(alpha_std + beta_std*x_std,
                 sigma_std);
}
\end{lstlisting}
}
\begin{itemize}
\item $\beta_{\mathrm{std}}$ measures change in mean response SDs
  per predictor SD. $\sigma_{\mathrm{std}}$ is residual SD in response SDs.
\item These are illustrative priors on the standardized scale.
  They need not match the earlier original-scale priors.
\end{itemize}
\end{frame}

\begin{frame}[fragile]
\frametitle{Standardized regression: original units}

Expanding $y=m_y+s_y\{\alpha_{\mathrm{std}}
+\beta_{\mathrm{std}}(x-m_x)/s_x+\text{error}\}$ gives
\[
\beta=\frac{s_y}{s_x}\beta_{\mathrm{std}},\qquad
\alpha=m_y+s_y\alpha_{\mathrm{std}}-\beta m_x,\qquad
\sigma=s_y\sigma_{\mathrm{std}}.
\]
{\small
\begin{lstlisting}[language=Stan]
generated quantities {
  real beta = beta_std * sy / sx;
  real alpha = my + sy*alpha_std - beta*mx;
  real sigma = sy*sigma_std;
  real mu_pred = alpha + beta*xpred;
  real y_pred = normal_rng(mu_pred, sigma);
}
\end{lstlisting}
}
\begin{itemize}
\item Transform \uured{every posterior draw}, including the residual SD.
\item \texttt{xpred}, \texttt{mu\_pred} and \texttt{y\_pred} are now
  in original units. Do not standardize the prediction point again.
\end{itemize}
\end{frame}

\frametitle{Extreme value analysis}
  {\footnotesize
  \begin{lstlisting}[language=Stan]
data {
  int<lower=1> N;
  vector<lower=0>[N] y;
  int<lower=0> Nt;
  vector<lower=0>[Nt] yt;
}
transformed data {
  real ymax;
  ymax = max(y);
  if (ymax <= 0)
    reject("At least one y must be positive");
}
parameters {
  real<lower=0> sigma;
  real<lower=-sigma/ymax> k;
}
model {
  y ~ gpareto(k, sigma);
}
generated quantities {
  vector[Nt] predccdf;
  predccdf = gpareto_ccdf(yt,k,sigma);
}
  \end{lstlisting}
}
\end{frame}

\begin{frame}[fragile]

\frametitle{Functions: generalized Pareto log density}
  {\footnotesize
  \begin{lstlisting}[language=Stan]
functions {
  real gpareto_lpdf(vector y,
                    real k, real sigma) {
    int N = num_elements(y);
    if (sigma <= 0)
      reject("sigma must be positive");
    if (N == 0) return 0;
    if (min(y) < 0 ||
        (k < 0 && max(y) >= -sigma/k))
      return negative_infinity();
    if (k == 0)
      return -sum(y)/sigma - N*log(sigma);
    return -(1 + k)*sum(log1p(k*y/sigma))/k
           - N*log(sigma);
  }
  // Add the survival function shown next here.
}
  \end{lstlisting}
}
\begin{itemize}
\item The density is zero outside the support, so its log is $-\infty$. For $k=0$, the model is exponential.
\end{itemize}
\end{frame}

\begin{frame}[fragile]
\frametitle{Functions: generalized Pareto survival}
  {\footnotesize
  \begin{lstlisting}[language=Stan]
// In the same functions block:
vector gpareto_ccdf(vector y,
                    real k, real sigma) {
  vector[num_elements(y)] s;
  if (sigma <= 0)
    reject("sigma must be positive");
  for (n in 1:num_elements(y)) {
    if (y[n] <= 0)
      s[n] = 1;
    else if (k < 0 && y[n] >= -sigma/k)
      s[n] = 0;
    else if (k == 0)
      s[n] = exp(-y[n]/sigma);
    else
      s[n] = exp(-log1p(k*y[n]/sigma)/k);
  }
  return s;
}
  \end{lstlisting}
}
\begin{itemize}
\item Returns $P(Y>y_n\mid k,\sigma)$, not its logarithm.
\item For $k<0$, this probability is zero at and above the upper endpoint $-\sigma/k$.
\end{itemize}
\end{frame}
