\begin{frame}
\frametitle{Notation for hierarchical models}

  \begin{itemize}
  \item $j=1,\ldots,J$: groups; $i=1,\ldots,n_j$: observations in group $j$
  \item $y_{ij}$: individual observation; $y_j$: group data or a group summary;
    $y=(y_1,\ldots,y_J)$
  \item $\theta_j$: group-specific parameter;
    $\theta=(\theta_1,\ldots,\theta_J)$
  \item $\phi$: population hyperparameters
  \end{itemize}
  \medskip
  \textbf{For normal hierarchies:}
  \begin{itemize}
  \item $\mu$: population mean; $\tau$: between-group SD
  \item $\sigma$: within-group SD of individual observations
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Hierarchical model: assumptions}

  \[
    \begin{xy}
      \xymatrix@R=5mm@C=8mm{
        & \phi \ar[dl] \ar[d] \ar[drr] & & \\
        \theta_1\ar[d] & \theta_2\ar[d] & \cdots & \theta_J\ar[d] \\
        y_1 & y_2 & & y_J
      }
    \end{xy}
  \]
  \begin{itemize}
  \item Given $\phi$, the $\theta_j$ are independent with a common
    population distribution $p(\theta_j\mid\phi)$
  \item<2-> Given $\theta$, groups are independent, and $y_j$
    depends only on its own parameter $\theta_j$
  \item<2-> The hyperprior $p(\phi)$ describes uncertainty about
    the population distribution
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Hierarchical model: joint posterior}

  Bayes' rule and the conditional-independence assumptions give
  \begin{align*}
    p(\theta,\phi\mid y)
      &\propto p(y\mid\theta)\,p(\theta\mid\phi)\,p(\phi)\\
      &\propto p(\phi)\prod_{j=1}^J
        \left[p(\theta_j\mid\phi)\,p(y_j\mid\theta_j)\right].
  \end{align*}
  \begin{itemize}
  \item One hyperprior; each group contributes a population factor
    and a likelihood
  \item<2-> Given $\phi$ and $y$, the group parameters are independent
  \item<2-> Averaging over the shared, uncertain $\phi$ generally
    makes the $\theta_j$ dependent in the posterior
  \item<2-> Each group's data inform $\phi$, so other groups can
    inform its estimate of $\theta_j$
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Beta-binomial conditional posterior}

  For fixed $\alpha,\beta>0$, the binomial likelihood gives
  \begin{align*}
    p(\theta_j\mid\alpha,\beta,y_j)
      &\propto \theta_j^{y_j}(1-\theta_j)^{n_j-y_j}
        \theta_j^{\alpha-1}(1-\theta_j)^{\beta-1}\\
      &\propto \theta_j^{\alpha+y_j-1}
        (1-\theta_j)^{\beta+n_j-y_j-1}.
  \end{align*}
  Thus
  \[
    \theta_j\mid\alpha,\beta,y_j
      \sim\operatorname{Beta}(\alpha+y_j,\beta+n_j-y_j).
  \]
  \begin{itemize}
  \item The population distribution supplies $\alpha,\beta$
  \item Group $j$ supplies $y_j$ successes and $n_j-y_j$ failures
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Beta-binomial shrinkage}

  Write the population mean and concentration as
  \[
    m=\frac{\alpha}{\alpha+\beta},\qquad
    \kappa=\alpha+\beta.
  \]
  For $n_j>0$,
  \begin{align*}
    \operatorname{E}(\theta_j\mid\alpha,\beta,y_j)
      &=\frac{\alpha+y_j}{\alpha+\beta+n_j}\\
      &=\underbrace{\frac{n_j}{n_j+\kappa}}_{\text{data weight}}
        \frac{y_j}{n_j}
        +\underbrace{\frac{\kappa}{n_j+\kappa}}_{\text{population weight}}m.
  \end{align*}
  \begin{itemize}
  \item The conditional posterior mean lies between $y_j/n_j$ and $m$
  \item With $\alpha,\beta$ fixed, smaller groups receive more
    population weight
  \item Full Bayes also averages over uncertainty in $\alpha,\beta$
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Hierarchical binomial model: rats}

  \begin{itemize}
  \item Population prior $\Beta(\theta_j|{\color{red}\alpha,\beta})$
  \item Hyperprior $p({\color{red}\alpha,\beta})$?
    \begin{itemize}
    \item ${\color{red}\alpha,\beta}$ both affect the location and scale
    \item BDA3 (pp. 110--111) uses, for $\alpha,\beta>0$,
      $p({\color{red}\alpha,\beta})\propto({\color{red}\alpha}+{\color{red}\beta})^{-5/2}$
    \item This hyperprior is \textbf{improper}: its integral is infinite
    \item BDA shows that the posterior is proper for these rat data;
      posterior propriety must be checked when using an improper prior
    \item An example-specific choice, not a general default
    \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Hierarchical inference with posterior draws}

  For $s=1,\ldots,S$:
  \begin{enumerate}
  \item Draw $(\alpha^{(s)},\beta^{(s)})\sim p(\alpha,\beta\mid y)$
  \item For every group $j$, draw independently conditional on this pair:
    \[
      \theta_j^{(s)}\sim\operatorname{Beta}
        (\alpha^{(s)}+y_j,\beta^{(s)}+n_j-y_j)
    \]
  \end{enumerate}
  {\small\textit{Stan note:} If $\theta_j$ is a model parameter,
    Stan returns its posterior draws directly.}

  \textbf{Share one hyperparameter draw across all groups at each step.}
  \[
    \operatorname{E}[h(\theta_j)\mid y]
      =\int h(\theta_j)p(\theta_j\mid y)\,d\theta_j
      \approx\frac{1}{S}\sum_{s=1}^S h(\theta_j^{(s)}).
  \]
  Choosing $h(\theta)=\theta$ estimates a mean;
  $h(\theta)=\mathbf{1}(\theta>c)$ estimates a probability.
\end{frame}

\begin{frame}
\frametitle{Factory model: a complete hierarchy}

  One illustrative choice of \textbf{proper, independent priors} is
  \[
    \begin{gathered}
      \mu\sim\N(m_0,s_0^2),\\
      \tau\sim\operatorname{HalfNormal}(0,A_\tau),\qquad
      \sigma\sim\operatorname{HalfNormal}(0,A_\sigma).
    \end{gathered}
  \]
  The common-variance factory model then has
  \begin{align*}
    \theta_j\mid\mu,\tau&\overset{\mathrm{iid}}{\sim}\N(\mu,\tau^2),
      &&j=1,\ldots,J,\\
    y_{ij}\mid\theta_j,\sigma&\sim\N(\theta_j,\sigma^2),
      &&i=1,\ldots,n_j.
  \end{align*}
  \begin{itemize}
  \item $\operatorname{HalfNormal}(0,A)$: a $\N(0,A^2)$ restricted to
    $\mathbb{R}^+=(0,\infty)$
  \item Observations are independent given the group means and $\sigma$
  \item Fix $m_0,s_0,A_\tau,A_\sigma$ using subject knowledge (e.g. prior predictive distribution)
%  \item Here $\N$ uses variance as its second argument;
%    $A_\tau,A_\sigma$ are half-normal scale parameters
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{A half-normal prior for a standard deviation}

  A simple way to restrict $\tau$ to positive values is
  \[
    Z\sim\N(0,A_\tau^2),\qquad \tau=|Z|.
  \]
  This defines $\tau\sim\operatorname{HalfNormal}(0,A_\tau)$ on
  $\mathbb{R}^+=(0,\infty)$, with
  \[
    \Pr(\tau<1.96A_\tau)\approx0.95.
  \]
  \begin{itemize}
  \item The 0 denotes the mean of the underlying normal $Z$, not of $\tau$
  \item $A_\tau$ has the same units as the outcome
  \item Example: to put 95\% prior probability below a between-machine
    SD of 20 quality units, choose $A_\tau\approx20/1.96=10.2$
  \item Choose $A_\sigma$ separately for variation within a machine
  \item These scales are illustrative, not universal defaults
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{The prior on variation affects pooling}

  \begin{itemize}
  \item Small $\tau$: machine means lie close to $\mu$, encouraging
    stronger pooling
  \item Large $\tau$: machine means can differ more, allowing less pooling
  \item With few groups, the data may give little information about
    $\tau$, so its prior can matter substantially
  \item Check prior predictive outcomes and compare plausible prior scales
  \item A uniform prior on $\tau$, on $\tau^2$, or on $\log\tau$
    gives different assumptions; ``flat'' does not mean neutral
  \end{itemize}
  \medskip
  {\small BDA3, Section 5.7 discusses priors for hierarchical variation.
    The half-normal specification here is one modelling choice.}
\end{frame}

\begin{frame}
\frametitle{Factory: observations and means}

  \begin{itemize}
  \item In the common-variance model, observations are independent
    given $\theta_j$ and $\sigma$:
    \[
      y_{ij}\mid\theta_j,\sigma\sim\N(\theta_j,\sigma^2),
      \qquad i=1,\ldots,n_j.
    \]
  \item Machine $j$'s sample mean and its sampling variance are
    \begin{align*}
      \bar y_{.j}&=\frac{1}{n_j}\sum_{i=1}^{n_j}y_{ij},\\
      \operatorname{Var}(\bar y_{.j}\mid\theta_j,\sigma)
        &=\frac{\sigma^2}{n_j}=s_j^2.
    \end{align*}
    SD of observations: $\sigma$; SE of the mean: $s_j=\sigma/\sqrt{n_j}$.
    \pause
  \item Thus $\bar y_{.j}\mid\theta_j,\sigma\sim\N(\theta_j,s_j^2)$.
    We condition on $\sigma$ here; full Bayes also learns $\sigma$
    from the data.
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Normal shrinkage: the weights}

  For fixed $\mu$, $\tau>0$ and $\sigma>0$, let $s_j=\sigma/\sqrt{n_j}$.
  \[
    \operatorname{E}(\theta_j\mid\mu,\tau,\sigma,y)
      =w_j\bar y_{.j}+(1-w_j)\mu,\qquad
    w_j=\frac{\tau^2}{\tau^2+s_j^2}.
  \]
  \begin{itemize}
  \item A precise machine mean (small $s_j$) receives more weight
  \item Small between-machine variation $\tau$ gives more weight to $\mu$
  \item As $\tau\to0$, the conditional mean approaches $\mu$;
    as $\tau\to\infty$, it approaches $\bar y_{.j}$
  \item In full Bayes, average over $p(\mu,\tau,\sigma\mid y)$
  \item Strong pooling makes the machine means similar;
    their common value can still be uncertain because $\mu$ is unknown
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Normal shrinkage: a numerical example}

  Suppose machine 1 has $\bar y_{.1}=28$ from $n_1=4$ observations.
  For illustration, \textbf{fix} $\mu=8$, $\tau=5$ and $\sigma=30$,
  so $s_1=30/\sqrt{4}=15$.
  \begin{align*}
    w_1&=\frac{5^2}{5^2+15^2}=\frac{25}{250}=0.10,\\
    m_1&=0.10\cdot28+0.90\cdot8=10,\\
    v_1&=\left(s_1^{-2}+\tau^{-2}\right)^{-1}
      =\left(\frac{1}{225}+\frac{1}{25}\right)^{-1}=22.5.
  \end{align*}
  Posterior precision = data precision + prior precision.

  The conditional posterior is
  $\theta_1\mid\mu,\tau,\sigma,y\sim\N(m_1,v_1)$:
  mean $10$, SD $\sqrt{22.5}\approx4.74$.
  \begin{itemize}
  \item The noisy machine mean receives 10\% of the weight
  \item The full factory posterior also averages over $\mu,\tau,\sigma$
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Pair discussion: precision and shrinkage}

  {\color{uured}\textbf{Discuss in pairs (1--2 minutes).}}

  Two machines have the same sample mean, different from $\mu$.
  They share a within-machine SD $\sigma$, but different sample sizes
  give different standard errors of their means:
  \[
    s_1=5,\qquad s_2=20.
  \]
  Hold $\mu$, $\tau>0$ and $\sigma>0$ fixed.

  \textbf{Which machine's estimate is shrunk more toward $\mu$? Why?}

  \uncover<2->{
    \medskip
    \textbf{Answer:} Machine 2. Its larger standard error means less
    precise evidence, so its sample mean receives less weight and
    $\mu$ receives more weight.
  }
\end{frame}

\begin{frame}
\frametitle{Prediction for an existing machine}

  In the common-variance model, distinguish the mean $\theta_j$
  from a future observation $\tilde y_j$:
  \[
    p(\tilde y_j\mid y)=\int\!\int
      p(\tilde y_j\mid\theta_j,\sigma)
      p(\theta_j,\sigma\mid y)\,d\theta_j\,d\sigma.
  \]
  For each posterior draw $s$:
  \begin{enumerate}
  \item Take $(\theta_j^{(s)},\sigma^{(s)})$ from the
    \textbf{same joint posterior draw}
  \item Draw $\tilde y_j^{(s)}\sim
    \N(\theta_j^{(s)},(\sigma^{(s)})^2)$
  \end{enumerate}
  \begin{itemize}
  \item The $\theta_j^{(s)}$ summarize uncertainty about the machine's mean
  \item The $\tilde y_j^{(s)}$ also include variation in a new observation
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Prediction for a new similar machine}

  With no observations yet from the new machine,
  \begin{align*}
    p(\theta_{\mathrm{new}}\mid y)
      &=\int\!\int p(\theta_{\mathrm{new}}\mid\mu,\tau)
        p(\mu,\tau\mid y)\,d\mu\,d\tau,\\
    p(\tilde y_{\mathrm{new}}\mid y)
      &=\int\!\int p(\tilde y_{\mathrm{new}}\mid\theta_{\mathrm{new}},\sigma)
        p(\theta_{\mathrm{new}},\sigma\mid y)\,d\theta_{\mathrm{new}}\,d\sigma.
  \end{align*}
  For each posterior draw $s$:
  \begin{enumerate}
  \item Take $(\mu^{(s)},\tau^{(s)},\sigma^{(s)})$ jointly from
    $p(\mu,\tau,\sigma\mid y)$
  \item Draw the new mean:
    $\theta_{\mathrm{new}}^{(s)}\sim\N(\mu^{(s)},(\tau^{(s)})^2)$
  \item For a new observation, also draw
    $\tilde y_{\mathrm{new}}^{(s)}\sim
      \N(\theta_{\mathrm{new}}^{(s)},(\sigma^{(s)})^2)$
  \end{enumerate}
  Predicting a new mean stops at step 2; predicting an observation
  includes step 3.
\end{frame}

\begin{frame}
\frametitle{Pair discussion: a new machine}

  {\color{uured}\textbf{Discuss in pairs (1--2 minutes).}}

  We have data from six machines and want to predict for a new,
  similar machine 7.
  \begin{itemize}
  \item Its mean quality, $\theta_7$
  \item The quality of its next item, $\tilde y_7$
  \end{itemize}
  \textbf{How do the simulation steps differ? What additional
    uncertainty enters the second prediction?}

  \uncover<2->{
    \medskip
    \textbf{Answer:} Both start by drawing a new machine mean using
    posterior draws of the population parameters. Predicting an item
    also adds within-machine variation:
    \[
      \tilde y_7^{(s)}\sim\N(\theta_7^{(s)},(\sigma^{(s)})^2).
    \]
  }
\end{frame}

\section{Exchangeability}
\frame{\sectionpage}

\begin{frame}
\frametitle{Independence and exchangeability}

  \begin{enumerate}
  \item \textbf{Independence:} knowing one outcome does not change the
    distribution of another.
  \item \textbf{Exchangeability:} swapping the labels or positions of
    observations leaves their joint distribution unchanged.
  \end{enumerate}
\end{frame}

% In other words exchangeability is not a fundamental property of a system being observed but rather a consequence of how we choose to observe that system!

% In any principled modeling application we have to remain diligent in recognizing that exchangeability is an assumption, and one that cannot be made without careful consideration. Knowledge about the source of heterogeneity can drastically improve inferences, and we should consider exploiting that information whenever possible. Only when we don't have that knowledge, or don't have the resources to properly exploit it, will exchangeability be a reasonable assumption.


\begin{frame}

\frametitle{Exchangeability}

  \begin{itemize}
  \item Motivates a \uured{symmetric joint model}
    \begin{itemize}
    \item for observations whose labels carry no relevant information
    \item for group parameters that we treat symmetrically a priori
    \end{itemize}
  \item Less restrictive than \textbf{IID}
    (independent and identically distributed)
  \item IID implies exchangeability, but exchangeability need not
    imply independence
  \item Independence alone is not enough: the distributions may differ
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Exchangeability}

  \begin{itemize}
  \item \textit{Exchangeability} \\ Random variables
    $\theta_1,\ldots,\theta_J$ (or $y_1,\ldots,y_J$) are
    \uured{exchangeable} if the joint distribution $p$ is invariant to the
    permutation of indices\\
    i.e
    \begin{equation*}
      p(\theta_1,\theta_2,\theta_3) = p(\theta_{\pi(1)},\theta_{\pi(2)},\theta_{\pi(3)})
    \end{equation*}
    for any permutation $\pi$ of the indices. E.g.
    \begin{equation*}
      p(\theta_1,\theta_2,\theta_3) = p(\theta_2,\theta_3,\theta_1)
    \end{equation*}
\pause
  \uured{Can we come up with a situation where this doesn't hold?} \\ \pause
  E.g. when we have a trend over indices (like in time series data)
\pause
  \item Exchangeability implies symmetry: \\If there is no information
    which can be used \textit{a priori} to separate $\theta_j$ from
    each other, we can assume exchangeability. ("Ignorance implies exchangeability")

  \end{itemize}
\end{frame}


\begin{frame}

\frametitle{Exchangeability: de Finetti's theorem}

  \begin{itemize}
  \item \textbf{Bernoulli case:} let $(X_i)_{i=1}^{\infty}$ be an
    infinite exchangeable sequence with $X_i\in\{0,1\}$
  \item There is a random $\theta\in[0,1]$ such that the $X_i$ are
    \uured{conditionally IID}:
    \[
      X_i\mid\theta\overset{\mathrm{iid}}{\sim}
        \operatorname{Bernoulli}(\theta)
    \]
  \item If $\theta$ has a density $p(\theta)$, then for every finite $J$,
    \[
      p(x_1,\ldots,x_J)=\int_0^1
        \left[\prod_{i=1}^J\theta^{x_i}(1-\theta)^{1-x_i}\right]
        p(\theta)\,d\theta
    \]
  \item The theorem does not choose the mixing distribution:
    a Beta prior is a modelling choice, not a consequence
  \end{itemize}
\end{frame}



\begin{frame}

\frametitle{De Finetti's takeaway}

  \textbf{Exchangeability provides a justification for modelling
    observations as independent conditional on a shared unknown
    parameter, even though they need not be independent marginally.}

  \begin{enumerate}
  \item \textbf{Usually, we start with a Bayesian model:} an unknown
    coin bias $\theta$, a prior for $\theta$, and independent flips
    given $\theta$.
  \item \textbf{De Finetti starts with symmetry:} the order of the
    flips carries no information.
  \item For an \textbf{infinite exchangeable Bernoulli sequence}, the
    theorem says its distribution must admit exactly that Bayesian
    representation: a distribution over $\theta$ and IID flips
    conditional on $\theta$.
  \end{enumerate}

\end{frame}

\begin{frame}
\frametitle{Pair discussion: learning about the coin}

  {\color{uured}\textbf{Discuss in pairs (1--2 minutes).}}

  The coin's bias has prior $\theta\sim\operatorname{Beta}(1,1)$,
  and the flips are independent given $\theta$.

  \textbf{Why does observing heads change our prediction for the next
    flip? Does this contradict conditional independence?}

  \uncover<2->{
    \medskip
    \textbf{Answer:} We learn about the shared unknown bias.
    After observing heads, $\theta\mid\text{heads}\sim\operatorname{Beta}(2,1)$,
    changing the predictive probability from $1/2$ to $2/3$.

    The coin has not changed. Independence holds when $\theta$ is
    fixed; averaging over an uncertain $\theta$ creates dependence.
  }
\end{frame}

\begin{frame}

\frametitle{Coin flips: exchangeable but dependent}

  Let $\theta\sim\operatorname{Beta}(1,1)$ and, given $\theta$,
  let the flips be IID Bernoulli$(\theta)$.
  Write $H_i$ for heads on flip $i$.
  \[
    \Pr(H_1)=\operatorname{E}(\theta)=\frac12.
  \]
  \pause
  After observing $H_1$,
  \[
    \theta\mid H_1\sim\operatorname{Beta}(2,1),\qquad
    \Pr(H_2\mid H_1)=\operatorname{E}(\theta\mid H_1)=\frac23.
  \]
  \begin{itemize}
  \item Marginally $\Pr(H_2)=1/2$, so the flips are dependent
  \item Their joint distribution remains unchanged when we swap the flips
  \item Observing heads updates the shared unknown bias;
    one toss does not physically affect the next
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Exchangeability}

  \begin{itemize}
  \item Exchangeability does not require identical experiment probabilities
  \item Suppose experiments came from two laboratories with different
    conditions, but their laboratory memberships are unknown
  \item If we treat these unknown memberships symmetrically across
    experiments, we can model the probabilities $\theta_j$ as
    \textbf{exchangeable a priori}
  \item A model with unknown laboratory membership can represent
    clustering among experiments from the same laboratory
  \item This concerns $\theta_j$, not the counts $y_j$:
    different sample sizes $n_j$ can make the counts non-exchangeable
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Exchangeability and additional information}

  \begin{itemize}
  \item Bioassay: doses $x_j$ and group sizes $n_j$ are fixed
    design information; $y_j$ counts deaths
  \item<2-> Given $\alpha,\beta$, the counts are independent, but
    generally not identically distributed:
    \[
      y_j\mid\alpha,\beta,n_j,x_j\sim\operatorname{Binomial}(n_j,\pi_j),
      \qquad j=1,\ldots,J
    \]
    where $\operatorname{logit}(\pi_j)=\alpha+\beta x_j$.
    \[
      p(\alpha,\beta\mid y,n,x)\propto p(\alpha,\beta)
        \prod_{j=1}^J p(y_j\mid\alpha,\beta,n_j,x_j)
    \]
  \item<3-> Reordering complete records $(x_j,n_j,y_j)$ leaves the
    likelihood unchanged; swapping only counts between doses
    generally changes it
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Hierarchical exchangeability}

  \begin{itemize}
  \item Rat example: experiments $j=1,\ldots,J$, each with $n_j$ rats
  \item<2-> Within experiment $j$, outcomes are IID
    Bernoulli$(\theta_j)$ given $\theta_j$; they need not be
    exchangeable across experiments
  \item<3-> The group probabilities $\theta_j$ are modelled
    exchangeably a priori
  \item<4-> Counts $y_j\mid\theta_j,n_j\sim\operatorname{Binomial}(n_j,\theta_j)$
    need not be exchangeable when $n_j$ differs
  \item<4-> This assumes comparable experiments; grouping alone
    does not justify a shared population distribution
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Partial or conditional exchangeability}

\begin{itemize}
  \item \textbf{Conditional exchangeability}
    \begin{itemize}
    \item Responses with the same covariates may be modelled
      exchangeably; different covariates may imply different distributions
    \item For fixed covariates, use a conditional model
      $p(y\mid x)$; adding $x_i$ does not make
      $(x_i,y_i)$ exchangeable
    \end{itemize}
  \item<2-> \textbf{Partial exchangeability}
    \begin{itemize}
    \item Symmetry within specified groups, not under arbitrary
      exchanges across groups
    \item The models here additionally assume exchangeable group
      parameters, possibly given group-level covariates
    \end{itemize}
\end{itemize}
\end{frame}

\section{Computational aspects}
\frame{\sectionpage}

\begin{frame}
\frametitle{Funnel geometry: prior and posterior}

  \begin{itemize}
  \item Consider the \textbf{prior hierarchy}, before observing data:
  \[
    \tau\sim p(\tau),\quad \tau>0,\qquad
    \theta_j\mid\tau\sim\mathcal{N}(0,\tau)\quad\text{(SD form)}
  \]
  \item Small $\tau$: group effects lie near 0;
    large $\tau$: group effects are more spread out
  \item If plausible $\tau$ values range from near zero to large values,
    the joint prior has a narrow neck and a broad region: a funnel
  \item<2-> With weak information about group effects, the
    \textbf{posterior can retain} this funnel geometry
  \item<2-> Not every hierarchical posterior is funnel-shaped:
    the likelihood also matters
  \end{itemize}
\end{frame}


\begin{frame}
\frametitle{Why is a funnel problematic?}

  \begin{itemize}
  \item Changing curvature makes it hard for HMC to explore both
    the narrow and wide regions efficiently
  \item Post-warmup divergences warn of potentially unreliable
    exploration and biased posterior estimates
  \item Good $\widehat R$ and ESS do not cancel divergence warnings
  \end{itemize}
  \begin{center}
    \includegraphics[width=5.7cm]{figs/diver.png}\\
    {\tiny Betancourt (2020)\quad
      \href{https://mc-stan.org/learn-stan/diagnostics-warnings.html}{Stan diagnostics}\quad
      \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=HamiltonianMC&target=funnel}{HMC demo}}
    \\{\scriptsize In this figure, $\phi$ denotes one group effect $\theta_j$.}
  \end{center}

\end{frame}


\begin{frame}
\frametitle{Handling the funnel}

  \begin{itemize}
  \item Raising \texttt{adapt\_delta} targets smaller steps:
    it can help, but does not guarantee reliable inference
  \item \textbf{Centered:} often better with strong group information
  \[
    \theta_j\sim N(\mu,\tau)\quad\text{(SD form)}
  \]
  \item<2-> \textbf{Non-centered:} often helps weakly informed
    group effects
  \uncover<2->{
  \[
    \eta_j\sim N(0,1),\qquad \theta_j=\mu+\tau\eta_j
  \]
  }
  \end{itemize}
  \begin{center}
    \includegraphics[width=6cm]{figs/centered_noncentered.png}\\
    {\tiny Betancourt (2020)\quad
      \href{https://mc-stan.org/docs/stan-users-guide/efficiency-tuning.html}{Stan: parameterization guidance}}
  \end{center}



\end{frame}

\begin{frame}
\frametitle{Why non-centering preserves the model}

  Start with $\eta_j\overset{\mathrm{iid}}{\sim}\N(0,1)$,
  independent of the hyperparameters a priori, and set
  \[
    \theta_j=\mu+\tau\eta_j,\qquad \tau>0.
  \]
  Conditional on $\mu,\tau$, an affine transformation of a normal
  variable is normal, with
  \begin{align*}
    \operatorname{E}(\theta_j\mid\mu,\tau)
      &=\mu+\tau\operatorname{E}(\eta_j)=\mu,\\
    \operatorname{Var}(\theta_j\mid\mu,\tau)
      &=\tau^2\operatorname{Var}(\eta_j)=\tau^2.
  \end{align*}
  Thus the $\theta_j$ are conditionally independent normals with
  mean $\mu$ and SD $\tau$, as in the centered hierarchy.

  Keep the same hyperpriors and likelihood, evaluated at
  $\theta_j=\mu+\tau\eta_j$. Only the sampling coordinates change.
\end{frame}

\begin{frame}
\frametitle{Normal shrinkage: completing the square}

  For fixed $\mu$ and $\tau>0$, with known sampling SD $\sigma_j>0$,
  \[
    y_j\mid\theta_j\sim\N(\theta_j,\sigma_j^2),\qquad
    \theta_j\mid\mu,\tau\sim\N(\mu,\tau^2).
  \]
  Here $\N$ uses variance as its second argument.
  \begin{align*}
    p(\theta_j\mid\mu,\tau,y)
      &\propto\exp\!\left[-\frac12\left\{
        \frac{(y_j-\theta_j)^2}{\sigma_j^2}
        +\frac{(\theta_j-\mu)^2}{\tau^2}\right\}\right]\\
      &\propto\exp\!\left[-\frac{(\theta_j-m_j)^2}{2v_j}\right],
  \end{align*}
  where completing the square gives
  \[
    v_j=\left(\frac{1}{\sigma_j^2}+\frac{1}{\tau^2}\right)^{-1},
    \qquad m_j=v_j\left(\frac{y_j}{\sigma_j^2}+\frac{\mu}{\tau^2}\right).
  \]
  Thus $\theta_j\mid\mu,\tau,y\sim\N(m_j,v_j)$.
\end{frame}
