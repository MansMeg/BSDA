\begin{frame}

\frametitle{Hierarchical model}

  \begin{itemize}
  \item Example: Treatment effectiveness
    \begin{itemize}
    \item Hospital $j$: survival probability $\theta_j$
    \item Patient $i$'s survival outcome: $y_{ij}$\\[1mm]
        \begin{xy}
          \xymatrix{ \theta_1\ar[d] & \theta_2\ar[d] &
            \cdots & \theta_J\ar[d] \\
            y_{i1} & y_{i2} & & y_{iJ} }
        \end{xy}
        \pause
      \item Model the $\theta_j$ as similar, not identical\\[1mm]
        \begin{xy}
          \xymatrix{
            & \phi \ar[dl] \ar[d] \ar[drr] & & &  \\
            \theta_1\ar[d] & \theta_2\ar[d] &
            \cdots & \theta_J\ar[d] \\
            y_{i1} & y_{i2} & & y_{iJ} }
        \end{xy}
      \item A shared population distribution has hyperparameters $\phi$
      \item Both $\theta_j$ and $\phi$ are unknown
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Notation for hierarchical models}

  \begin{itemize}
  \item $j=1,\ldots,J$: groups; $i=1,\ldots,n_j$: observations in group $j$
  \item $y_{ij}$: individual observation; $y_j$: group data or a group summary;
    $y=(y_1,\ldots,y_J)$
  \item $\theta_j$: group-specific parameter;
    $\theta=(\theta_1,\ldots,\theta_J)$
  \item $\phi$: population hyperparameters
  \end{itemize}
  \medskip
  \textbf{For normal hierarchies:}
  \begin{itemize}
  \item $\mu$: population mean; $\tau$: between-group SD
  \item $\sigma$: within-group SD of individual observations
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Hierarchical model: assumptions}

  \[
    \begin{xy}
      \xymatrix@R=5mm@C=8mm{
        & \phi \ar[dl] \ar[d] \ar[drr] & & \\
        \theta_1\ar[d] & \theta_2\ar[d] & \cdots & \theta_J\ar[d] \\
        y_1 & y_2 & & y_J
      }
    \end{xy}
  \]
  \begin{itemize}
  \item Given $\phi$, the $\theta_j$ are independent with a common
    population distribution $p(\theta_j\mid\phi)$
  \item<2-> Given $\theta$, groups are independent, and $y_j$
    depends only on its own parameter $\theta_j$
  \item<2-> The hyperprior $p(\phi)$ describes uncertainty about
    the population distribution
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Hierarchical model: joint posterior}

  Bayes' rule and the conditional-independence assumptions give
  \begin{align*}
    p(\theta,\phi\mid y)
      &\propto p(y\mid\theta)\,p(\theta\mid\phi)\,p(\phi)\\
      &\propto p(\phi)\prod_{j=1}^J
        \left[p(\theta_j\mid\phi)\,p(y_j\mid\theta_j)\right].
  \end{align*}
  \begin{itemize}
  \item One hyperprior; each group contributes a population factor
    and a likelihood
  \item<2-> Given $\phi$ and $y$, the group parameters are independent
  \item<2-> Averaging over the shared, uncertain $\phi$ generally
    makes the $\theta_j$ dependent in the posterior
  \item<2-> Each group's data inform $\phi$, so other groups can
    inform its estimate of $\theta_j$
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Hierarchical normal model for group means}

  \begin{itemize}
  \item Observations are independent given $\theta_j$; $\sigma^2$ is known
    \begin{equation*}
      y_{ij}|\theta_j \sim \N(\theta_j,\sigma^2), \quad
      i=1,\ldots,n_j; \quad j=1,\ldots,J
    \end{equation*}
  \item Group $j$ sample mean and its sampling variance
    \begin{align*}
      \bar{y}_{.j} & = \frac{1}{n_j}\sum_{i=1}^{n_j}y_{ij}\\
      \operatorname{Var}(\bar{y}_{.j}\mid\theta_j)
        & = \frac{\sigma^2}{n_j}=\sigma_j^2
    \end{align*}
    SD of observations: $\sigma$; SE of the mean: $\sigma_j=\sigma/\sqrt{n_j}$.
    \pause
  \item Thus $\bar{y}_{.j}\mid\theta_j \sim \N(\theta_j,\sigma_j^2)$.

    The same form applies to adjusted school effects $y_j$, using
    their reported standard errors $\sigma_j$.
  \end{itemize}
\end{frame}

\section{Exchangeability}
\frame{\sectionpage}
% In other words exchangeability is not a fundamental property of a system being observed but rather a consequence of how we choose to observe that system!

% In any principled modeling application we have to remain diligent in recognizing that exchangeability is an assumption, and one that cannot be made without careful consideration. Knowledge about the source of heterogeneity can drastically improve inferences, and we should consider exploiting that information whenever possible. Only when we don't have that knowledge, or don't have the resources to properly exploit it, will exchangeability be a reasonable assumption.


\begin{frame}

\frametitle{Exchangeability}

  \begin{itemize}
  \item Motivates a \uured{symmetric joint model}
    \begin{itemize}
    \item for observations whose labels carry no relevant information
    \item for group parameters that we treat symmetrically a priori
    \end{itemize}
  \item Less restrictive than \textbf{IID}
    (independent and identically distributed)
  \item IID implies exchangeability, but exchangeability need not
    imply independence
  \item Independence alone is not enough: the distributions may differ
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Exchangeability}

  \begin{itemize}
  \item \textit{Exchangeability} \\ Random variables
    $\theta_1,\ldots,\theta_J$ (or $y_1,\ldots,y_J$) are
    \uured{exchangeable} if the joint distribution $p$ is invariant to the
    permutation of indices\\
    i.e
    \begin{equation*}
      p(\theta_1,\theta_2,\theta_3) = p(\theta_{\pi(1)},\theta_{\pi(2)},\theta_{\pi(3)})
    \end{equation*}
    for any permutation $\pi$ of the indicies. E.g.
    \begin{equation*}
      p(\theta_1,\theta_2,\theta_3) = p(\theta_2,\theta_3,\theta_1)
    \end{equation*}
\pause
  \uured{Can we come up with a situation where this doesn't hold?} \\ \pause
  E.g. when we have a trend over indicies (like in time series data)
\pause
  \item Exchangeability implies symmetry: \\If there is no information
    which can be used \textit{a priori} to separate $\theta_j$ form
    each other, we can assume exchangeability. ("Ignorance implies exchangeability")

  \end{itemize}
\end{frame}


\begin{frame}

\frametitle{Exchangeability: de Finetti's theorem}

  \begin{itemize}
  \item \textbf{Bernoulli case:} let $(X_i)_{i=1}^{\infty}$ be an
    infinite exchangeable sequence with $X_i\in\{0,1\}$
  \item There is a random $\theta\in[0,1]$ such that the $X_i$ are
    \uured{conditionally IID}:
    \[
      X_i\mid\theta\overset{\mathrm{iid}}{\sim}
        \operatorname{Bernoulli}(\theta)
    \]
  \item If $\theta$ has a density $p(\theta)$, then for every finite $J$,
    \[
      p(x_1,\ldots,x_J)=\int_0^1
        \left[\prod_{i=1}^J\theta^{x_i}(1-\theta)^{1-x_i}\right]
        p(\theta)\,d\theta
    \]
  \item The theorem does not choose the mixing distribution:
    a Beta prior is a modelling choice, not a consequence
  \end{itemize}
\end{frame}



\begin{frame}

\frametitle{Example}

  \begin{itemize}
  \item Suppose we're observing coin flips, but we're not sure about the bias $\theta$
    \begin{itemize}
      \item We think the flips are 'the same kind of thing' (no reason to treat flip 1 differently from flip 2): \uured{Exchangeability}
      \item But we might not believe they are strictly independent because were uncertain about the bias: they share information through the unknown $\theta$
    \end{itemize}
    \pause
    \item Given $\theta$ the flips are i.i.d.
    \item Without conditioning on $\theta$, the flips are only exchangeable, not iid.
  \end{itemize}
\end{frame}




\begin{frame}

\frametitle{Exchangeability}

  \begin{itemize}
  \item Exchangeability does not mean that the results of the
    experiments could not be different
    \begin{itemize}
    \item e.g. if we know that the experiments have been in two
      different laboratories, and we know that the other laboratory
      has better conditions for the rats, but we do not know which
      experiments have been made in which laboratory
    \item a priori experiments are exchangeable
    \item model could have unknown parameter for the laboratory with a
      conditional prior for rats assumed to come form the same place (clustering model)
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Exchangeability and additional information}

  \begin{itemize}
  \item Bioassay: doses $x_j$ and group sizes $n_j$ are fixed
    design information; $y_j$ counts deaths
  \item<2-> Given $\alpha,\beta$, the counts are independent, but
    generally not identically distributed:
    \[
      y_j\mid\alpha,\beta,n_j,x_j\sim\operatorname{Binomial}(n_j,\pi_j),
      \qquad j=1,\ldots,J
    \]
    where $\operatorname{logit}(\pi_j)=\alpha+\beta x_j$.
    \[
      p(\alpha,\beta\mid y,n,x)\propto p(\alpha,\beta)
        \prod_{j=1}^J p(y_j\mid\alpha,\beta,n_j,x_j)
    \]
  \item<3-> Reordering complete records $(x_j,n_j,y_j)$ leaves the
    likelihood unchanged; swapping only counts between doses
    generally changes it
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Hierarchical exchangeability}

  \begin{itemize}
  \item Rat example: experiments $j=1,\ldots,J$, each with $n_j$ rats
  \item<2-> Within experiment $j$, outcomes are IID
    Bernoulli$(\theta_j)$ given $\theta_j$; they need not be
    exchangeable across experiments
  \item<3-> The group probabilities $\theta_j$ are modelled
    exchangeably a priori
  \item<4-> Counts $y_j\mid\theta_j,n_j\sim\operatorname{Binomial}(n_j,\theta_j)$
    need not be exchangeable when $n_j$ differs
  \item<4-> This assumes comparable experiments; grouping alone
    does not justify a shared population distribution
  \end{itemize}
\end{frame}

\begin{frame}

\frametitle{Partial or conditional exchangeability}

\begin{itemize}
  \item \textbf{Conditional exchangeability}
    \begin{itemize}
    \item Responses with the same covariates may be modelled
      exchangeably; different covariates may imply different distributions
    \item For fixed covariates, use a conditional model
      $p(y\mid x)$; adding $x_i$ does not make
      $(x_i,y_i)$ exchangeable
    \end{itemize}
  \item<2-> \textbf{Partial exchangeability}
    \begin{itemize}
    \item Symmetry within specified groups, not under arbitrary
      exchanges across groups
    \item The models here additionally assume exchangeable group
      parameters, possibly given group-level covariates
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Funnel geometry: prior and posterior}

  \begin{itemize}
  \item Consider the \textbf{prior hierarchy}, before observing data:
  \[
    \tau\sim p(\tau),\quad \tau>0,\qquad
    \theta_j\mid\tau\sim\mathcal{N}(0,\tau)\quad\text{(SD form)}
  \]
  \item Small $\tau$: group effects lie near 0;
    large $\tau$: group effects are more spread out
  \item If plausible $\tau$ values range from near zero to large values,
    the joint prior has a narrow neck and a broad region: a funnel
  \item<2-> With weak information about group effects, the
    \textbf{posterior can retain} this funnel geometry
  \item<2-> Not every hierarchical posterior is funnel-shaped:
    the likelihood also matters
  \end{itemize}
\end{frame}


\begin{frame}
\frametitle{Why is a funnel problematic?}

  \begin{itemize}
  \item Changing curvature makes it hard for HMC to explore both
    the narrow and wide regions efficiently
  \item Post-warmup divergences warn of potentially unreliable
    exploration and biased posterior estimates
  \item Good $\widehat R$ and ESS do not cancel divergence warnings
  \end{itemize}

      \href{https://mc-stan.org/learn-stan/diagnostics-warnings.html}{Stan diagnostics}\quad
      \href{https://chi-feng.github.io/mcmc-demo/app.html?algorithm=HamiltonianMC&target=funnel}{HMC demo}}
    \\{\scriptsize In this figure, $\phi$ denotes one group effect $\theta_j$.}
  \end{center}

\end{frame}


\begin{frame}
\frametitle{Handling the funnel}

  \begin{itemize}
  \item Raising \texttt{adapt\_delta} targets smaller steps:
    it can help, but does not guarantee reliable inference
  \item \textbf{Centered:} often better with strong group information
  \[
    \theta_j\sim N(\mu,\tau)\quad\text{(SD form)}
  \]
  \item<2-> \textbf{Non-centered:} often helps weakly informed
    group effects
  \uncover<2->{
  \[
    \eta_j\sim N(0,1),\qquad \theta_j=\mu+\tau\eta_j
  \]
  }
  \end{itemize}
