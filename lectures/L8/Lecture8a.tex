  \begin{itemize}
  \item<1-> Newcomb's speed of light measurements ($n=66$)
    \begin{itemize}
    \item $y_i\mid\mu,\sigma \sim \N(\mu,\sigma^2)$, conditionally independently
    \item improper prior $p(\mu,\log\sigma)\propto 1$, or $p(\mu,\sigma)\propto 1/\sigma$

    \item<3-> draw one pair $(\mu^{(s)},\sigma^{(s)})$ from $p(\mu,\sigma\mid y)$

      $y_i^{\mathrm{rep},(s)}\sim\N(\mu^{(s)},(\sigma^{(s)})^2)$, $i=1,\ldots,n$
    \item<5-> repeat both steps for $s=1,\ldots,S$: $S$ datasets of size $n$
    \item<6-> one replicated dataset:
    \uncover<6->{\includegraphics[width=6cm]{figs/light_ppc_1hist.pdf}}
      \end{itemize}
    \end{itemize}

\begin{frame}[fragile]

\frametitle{Choosing a discrepancy: ancillarity}

  \begin{itemize}
  \item<1-> A statistic is \uured{ancillary} if its sampling distribution
    does not depend on the unknown parameters.
  \item<2-> Example: in a Normal model with known $\sigma^2$, the sampling
    distribution of $s^2$ does not depend on the unknown $\mu$.
    If $\sigma^2$ is also unknown, $s^2$ is no longer ancillary.
  \item<3-> For checking, target features the model may miss, such as
    tails, skewness or dependence. A useful discrepancy need not be ancillary.
  \end{itemize}

\end{frame}

\begin{frame}[fragile]

\frametitle{Posterior predictive $p$-value}

  \begin{itemize}
  \item<1-> How often is the replicated discrepancy at least as large?
    {\small
    \begin{align*}
      p_B &= \Pr(T(y^{\rep},\theta)\geq T(y,\theta)\mid y)\\
          &= \iint I\{T(y^{\rep},\theta)\geq T(y,\theta)\}
             p(y^{\rep}\mid\theta)p(\theta\mid y)dy^{\rep}d\theta.
    \end{align*}
    }
  \item<2-> Draw $\theta^{(s)}\sim p(\theta\mid y)$, then
    $y^{\rep,(s)}\sim p(y^{\rep}\mid\theta^{(s)})$:
    \[
      \widehat p_B=\frac{1}{S}\sum_{s=1}^S
      I\{T(y^{\rep,(s)},\theta^{(s)})\geq T(y,\theta^{(s)})\}.
    \]
    Here $I\{\cdot\}$ is 1 if the condition holds, and 0 otherwise.
  \item<3-> Values near 0 or 1 can reveal a mismatch; a non-extreme value
    does not establish that the model fits well.
  \item<4-> Not the probability that the model is correct; not generally
    uniform under a correct model.
  \end{itemize}

\end{frame}

\subsection{Marginal Predictive Checking}
\frame{\subsectionpage}


\begin{frame}[fragile]

\frametitle{Marginal and CV predictive checking}

  \begin{itemize}
  \item For continuous observations, consider each predictive distribution:
    \[
      p_i=\Pr(y_i^{\rep}\leq y_i\mid y).
    \]
    The observation $y_i$ is also used to fit the model. These tail areas
    are not generally uniform, even for a correct model.
  \item<2-> Leave-one-out probability integral transform (LOO-PIT):
    \[
      u_i=\Pr(\tilde y_i\leq y_i\mid y_{-i})=F_{-i}(y_i).
    \]
    Here $F_{-i}$ is the predictive CDF fitted without $y_i$.
  \item<2-> For calibrated continuous predictions, approximate uniformity
    on $[0,1]$ is a useful benchmark, not an exact small-sample guarantee.
  \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Sensitivity analysis}

  \begin{itemize}
  \item How much different choices in model structure and priors affect the results
    \begin{itemize}
      \item<2-> test different models and priors
      \item<3-> alternatively combine different models to one model
        \begin{itemize}
        \item e.g. hierarchical model instead of separate and pooled
        \item e.g. Student's $t$ approaches a Gaussian as $\nu\to\infty$
      \end{itemize}
      \item<3-> robust models are good for testing sensitivity to ``outliers''
        \begin{itemize}
        \item e.g. $t$ instead of Gaussian
        \end{itemize}
    \end{itemize}
    \item<4-> Compare sensitivity of essential inference quantities
      \begin{itemize}
      \item extreme quantiles are often more sensitive than means and medians
      \item extrapolation is often more sensitive than interpolation
      \end{itemize}
    \end{itemize}

\end{frame}

\begin{frame}

\frametitle{Example: Exposure to air pollution}

  \begin{itemize}
  \item Three models:
  \begin{enumerate}
  \item Linear regression
  \[
  y_{ij} \sim N(\beta_0 + \beta_1 x_{ij}, \sigma^2)
  \]
  \item Hierarchical linear regression (WHO super regions)
  \item Hierarchical linear regression (clustered super regions)
  \[
  y_{ij} \sim N(\beta_0 + \beta_{0j} + (\beta_1 + \beta_{1j}) x_{ij}, \sigma^2)
  \]
  \[
  \beta_{0j}\sim N(0,\tau_0^2),\qquad \beta_{1j}\sim N(0,\tau_1^2)
  \]
  \end{enumerate}
  Here $y_{ij}$ is the natural log of $\text{PM}_{2.5}$ concentration,
  $x_{ij}$ is the natural log of the satellite estimate, and $j$ indexes regions.
  The model also requires priors
  for $\beta_0,\beta_1,\tau_0,\tau_1,\sigma$.
  \end{itemize}
\end{frame}


\begin{frame}

\frametitle{Air pollution: which priors change?}

  Both concentrations are measured in $\mu\mathrm{g}/\mathrm{m}^3$.
  We model their \uured{natural logarithms}: $y=\ln(c)$, $x=\ln(s)$.

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \uured{Vague priors}
      \begin{align*}
        \beta_0,\beta_1&\sim N(0,100^2),\\
        \tau_0^2,\tau_1^2,\sigma^2&\sim\mathrm{IG}(1,100).
      \end{align*}
    \end{column}
    \begin{column}{0.49\textwidth}
      \uured{Weakly informative}
      \begin{align*}
        \beta_0&\sim N(0,1),\\
        \beta_1&\sim N(1,1),\\
        \tau_0,\tau_1,\sigma&\sim N_+(0,1).
      \end{align*}
    \end{column}
  \end{columns}

  {\small $N$ uses variance as its second parameter.
  IG is inverse-gamma with shape and scale; $N_+$ is half-normal on positive values.}

  The tighter priors centre the calibration around intercept 0 and slope 1.

  {\scriptsize Priors follow the simulation code accompanying
  \href{https://github.com/jgabry/bayes-vis-paper}{Gabry et al. (2019)}.}
\end{frame}
