\section{Model assessment and selection}
\frame{\sectionpage}

\begin{frame}{Model checking and predictive performance}

  \begin{itemize}
  \item<1-> \uured{In L8a: What does the model miss?}\\
    Posterior predictive checks compare features of observed and replicated data.
  \item<2-> \uured{In L8b: How well does the model predict?}\\
    We assess predictions for the task of interest using held-out observations.
  \item<3-> \uured{The best candidate can still fit poorly.}\\
    All candidates may miss important features. Predictive comparisons
    complement model checks; they do not replace them.
  \end{itemize}

\end{frame}


\begin{frame}
\frametitle{Predictive performance}

\begin{itemize}
   \item<1-> Modeling complex phenomena with models that are simplified\\
   \uured{All models are wrong... but some are useful.}
  \item<2-> True predictive performance is found out by using it to \uured{make predictions} and comparing predictions to \uured{true observations}
    \begin{itemize}
      \item external validation
    \end{itemize}
  \item<3-> Expected predictive performance
    \begin{itemize}
      \item \uured{approximates} the external validation
    \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{Goal of model evaluation}

\begin{itemize}
\item Model choice is a (model-)decision-theoretic problem
\pause
\item Evaluate the \uured{utility} of a model $M$ for new \uured{unseen data} $\tilde{y}$:
\[
U_M = \int u_M(\tilde{y};y) p_\text{true}({\color{uured}\tilde{y}}) d{\color{uured}\tilde{y}}\,,
\]
where $\tilde y$ is an unseen observation and $y$ are the training data.
The utility $u_M(\tilde y;y)$ evaluates the prediction made by model $M$
after fitting it to $y$.
  \pause
  \item The expectation is with respect to $p_\text{true}$ ($f$ in BDA3)
  \pause
  \item Choose the model function to \uured{maximize our utility}
\end{itemize}
\end{frame}

\begin{frame}

\frametitle{Model choice utility}
  \begin{itemize}
  \item Application specific utility/cost functions are important
    \begin{itemize}
      \item eg. money, life years, quality adjusted life years, etc.
    \end{itemize}
  \pause
  \item General utility: \uured{overall in the goodness of the predictive distribution}
  \begin{itemize}
    \item we don't know (yet) the application specific utility\\
    then good information theoretically justified choice is log-score for model $M$
  \end{itemize}
      \begin{align*}
        \log p_M (y^\text{rep} | y)
      \end{align*}
    \pause
    \item We want the model that \uured{explain the data} the "best"
    % TODO: Increase this part on theoretical justification
\end{itemize}

\end{frame}

\subsection{Measures of predictive accuracy}
\frame{\subsectionpage}

\begin{frame}

\frametitle{Bayesian point prediction accuracy}

%  \item Scoring \uured{function}:
    \begin{itemize}
      \item Training residuals (the model is fitted to all of $y$)
      \begin{align*}
        e_i =  y_i - E(\tilde{y}_i|y)\,,
      \end{align*}
      where
      \begin{align*}
        E(\tilde{y}_i|y) = \int \tilde{y}_i p(\tilde{y}_i|y) d\tilde{y}_i\,,
      \end{align*}
      i.e. the \uured{expected predicted value}
      \pause
      \item Training mean squared error (MSE)
      \begin{align*}
        \text{MSE} = \frac{1}{n}\sum^n_i [y_i - E(\tilde{y}_i|y)]^2 \,.
      \end{align*}
      \pause
      \item But how do we evaluate the \uured{posterior predictive distribution}?
    \end{itemize}

\end{frame}


\begin{frame}
\frametitle{Probabilistic predictions: Scoring rules}

  \begin{itemize}
      \item A scoring rule evaluates a predictive distribution $q$ at an outcome $z$:
      \[
      S(q,z)\in\mathbb R\qquad\text{(higher is better).}
      \]
      Here $q$ is a probability density or probability mass function.
      \pause
      \item A \uured{proper} rule has its highest \emph{expected} score when the
        forecast matches the true distribution:
            \[
            \mathbb{E}_{Z\sim p_{\mathrm{true}}}[S(p_{\mathrm{true}},Z)]
            \geq \mathbb{E}_{Z\sim p_{\mathrm{true}}}[S(q,Z)].
            \]
        \pause
      \item A \uured{local} rule depends on the forecast $q$ only through its
        value $q(z)$ at the observed outcome.
  \end{itemize}

\end{frame}


\begin{frame}
\frametitle{Densities and probabilities}

  \begin{itemize}
  \item For a continuous outcome, $p(y_i\mid y)$ is a \uured{density}.
    The probability of exactly one value is zero; probabilities come
    from integrating the density over intervals.
  \item<2-> A density can exceed 1, so its logarithm can be positive.
    For example, a $\mathrm{Uniform}(0,0.5)$ density is 2 inside its support:
    $\log 2\approx0.69$.
  \item<3-> For a discrete outcome, $p(y_i\mid y)$ is a probability mass,
    between 0 and 1. Compare scores on the same outcome scale and units.
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Probabilistic predictions}

    \begin{itemize}
      \item The log score is local and proper: higher is better.
      \begin{align*}
         S(p(\cdot\mid y),\tilde y_i) = \log p(\tilde y_i\mid y).
      \end{align*}
%      proportional to MSE if constant variance
% TODO: Add example of different prediction normal distributions and the density
      \pause
      \item Evaluate the predictive density at an observed value $y_i$:
      \begin{align*}
        \log p(y_i\mid y)
        &= \log\int p(y_i\mid\theta)p(\theta\mid y)d\theta.
      \end{align*}
      This is a training score: $y_i$ was also used in fitting the model.
  \end{itemize}

\end{frame}

\begin{frame}
\frametitle{Pair discussion: log scores}

  {\color{uured}\textbf{Discuss in pairs (1--2 minutes).}}

  For a new Bernoulli outcome $\tilde y$, model A predicts
  $\Pr(\tilde y=1)=0.8$ and model B predicts $\Pr(\tilde y=1)=0.4$.

  \textbf{Which model has the higher log score if success occurs?
    What if failure occurs?}

  \uncover<2->{
    \medskip
    \textbf{Answer:} Using natural logarithms,
    \[
      \begin{array}{lcc}
        & \text{Model A} & \text{Model B} \\
        \text{Success }(\tilde y=1) & \log(0.8)\approx-0.22 & \log(0.4)\approx-0.92 \\
        \text{Failure }(\tilde y=0) & \log(0.2)\approx-1.61 & \log(0.6)\approx-0.51
      \end{array}
    \]
    A scores higher for success, B for failure. A confident prediction
    earns a better score when right and a worse score when wrong.
  }
\end{frame}

\begin{frame}
\frametitle{Probabilistic predictions}

    \begin{itemize}
      \item The log \uured{pointwise} predictive density (lppd):
      \begin{align*}
        \text{lppd} &= \sum_{i=1}^n\log p(y_i\mid y).
      \end{align*}
      This is not generally the joint log predictive density $\log p(y\mid y)$:
      integrating out shared parameters can induce dependence.
      \pause
      \item Monte Carlo approximation with $\theta^{(s)}\sim p(\theta\mid y)$:
      \begin{align*}
        \text{lppd} &\approx \sum_{i=1}^n\log\left(\frac{1}{S}
          \sum_{s=1}^S p(y_i\mid\theta^{(s)})\right).
      \end{align*}
  \end{itemize}

\end{frame}


\subsection{Model selection}
\frame{\subsectionpage}

\begin{frame}{Model selection}

\begin{itemize}
\item Evaluate model $M$ on unseen outcomes, one for each of $n$ prediction cases:
\[
\text{elpd}_M=\sum_{i=1}^n\int\log p_M(\tilde y_i\mid y)
  p_\text{true}(\tilde y_i)d\tilde y_i.
\]
Here $\tilde y_i$ is a future outcome for case $i$ and $y$ are the training data.
For regression, each prediction conditions on its covariates (not shown).
\item This is the \uured{total} expected log predictive density; divide by $n$
  for an average per observation.
\pause
\item The expectation is with respect to $p_\text{true}$
\pause
\item $p_\text{true}$ is (almost always) \uured{unknown}
\pause
\item The utility for each case is $\log p_M(\tilde y_i\mid y)$: higher is better.
\end{itemize}
\end{frame}

\section{Cross-validation}
\frame{\sectionpage}

\begin{frame}{Leave-one-out cross-validation (LOO-CV)}
\begin{itemize}
\item Our target is the total expected log predictive density:
\[
\text{elpd}_M=\sum_{i=1}^n\int\log p_M(\tilde y_i\mid y)
  p_\text{true}(\tilde y_i)d\tilde y_i.
\]
The future outcomes $\tilde y_i$ come from the true data-generating process;
$y$ are the observed training data.
\pause
\item How can we estimate these expectations without collecting new data?
\end{itemize}
\end{frame}


\begin{frame}{Leave-one-out cross-validation (LOO-CV)}

\begin{itemize}
\item Use held-out observations to estimate expected log predictive performance.
\pause
\item Hold out observation $i$ and try to predict $y_i$ based on $\mathbf{y}_{-i}$
%\pause
\item Estimation of $\text{elpd}_M$ using \uured{leave-one-out cross-validation}
\small
\begin{align*}
\text{elpd}_\text{loo} = & \sum^n_{i=1} \log p_M(y_i|\mathbf{y}_{-i}) \\
 = & \sum^n_{i=1} \log \int p_M(y_i|\theta) p(\theta|\mathbf{y}_{-i})d\theta
\end{align*}
\pause
\item \uured{Analogy:} replace expectations over future outcomes with held-out
  observations. Each fitted model uses $n-1$ observations.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{What does $p_{\mathrm{loo}}$ measure?}

  \[
    p_{\mathrm{loo}}=
    \underbrace{\sum_i\log p(y_i\mid y)}_{\mathrm{lppd}:\ \text{training fit}}
    -\underbrace{\sum_i\log p(y_i\mid y_{-i})}_{\mathrm{elpd}_{\mathrm{loo}}:\ \text{held-out fit}}.
  \]
  \begin{itemize}
  \item<1-> The gap measures how much better the model predicts observations
    it has already seen than observations left out of fitting.
  \item<2-> Often interpreted as an \uured{effective number of parameters}.
    It need not equal the number of coefficients, or even be an integer.
  \item<3-> In the coming example: $-26.8-(-29.5)=2.7$.
    Interpret this only after checking that the LOO computations are reliable.
  \end{itemize}
  {\scriptsize \href{https://mc-stan.org/loo/reference/loo-glossary.html}{loo glossary: effective number of parameters}.}
\end{frame}

\begin{frame}
  \frametitle{Importance sampling for LOO-CV}

   \begin{itemize}
   \item Let $f(\theta)=p(\tilde y_i\mid\theta)$, with proposal
     $g(\theta)=p(\theta\mid y)$ and target $q(\theta)=p(\theta\mid y_{-i})$.
   \pause
   \item Using draws $\theta^{(s)}\sim p(\theta\mid y)$:
     \begin{align*}
       p(\tilde y_i\mid y_{-i})\approx
       \frac{\sum_{s=1}^S p(\tilde y_i\mid\theta^{(s)})r_i^{(s)}}{\sum_{s=1}^S r_i^{(s)}},
     \end{align*}
     where the raw importance weights need only be proportional to
     \begin{align*}
       r_i^{(s)}\propto\frac{p(\theta^{(s)}\mid y_{-i})}{p(\theta^{(s)}\mid y)}.
     \end{align*}
   \end{itemize}

 \end{frame}

\begin{frame}
  \frametitle{Importance sampling for LOO-CV}

   \begin{itemize}
   \item For conditionally independent observations,
     $p(y\mid\theta)=\prod_j p(y_j\mid\theta)$:
     \begin{align*}
       \frac{p(\theta^{(s)}\mid y_{-i})}{p(\theta^{(s)}\mid y)}
       =\frac{p(y)}{p(y_{-i})}\frac{1}{p(y_i\mid\theta^{(s)})}
       \propto\frac{1}{\color{red}p(y_i\mid\theta^{(s)})}.
     \end{align*}
   \item Define $r_i^{(s)}=1/p(y_i\mid\theta^{(s)})$, so
     $\log r_i^{(s)}=-\log p(y_i\mid\theta^{(s)})$.
 \pause
   \item Evaluating at $\tilde y_i=y_i$ gives the raw IS estimate:
        \begin{align*}
       p(y_i\mid y_{-i})&\approx
       \frac{\sum_{s=1}^S p(y_i\mid\theta^{(s)})r_i^{(s)}}{\sum_{s=1}^S r_i^{(s)}}
       =\frac{S}{\sum_{s=1}^S\frac{1}{p(y_i\mid\theta^{(s)})}}.
     \end{align*}
     Each term in the numerator is 1, so the numerator is $S$.
%    \pause
%    \item What $\theta^{(s)}$ will give large weights?
% TODO: Add normal likelihood with known variance and compute the weights as an example.
   \end{itemize}
 \end{frame}

\begin{frame}
\frametitle{Pair discussion: LOO importance weights}

  {\color{uured}\textbf{Discuss in pairs (1--2 minutes).}}

  Three draws from the full posterior give likelihood values for $y_i$:
  \[
    p(y_i\mid\theta^{(1)})=0.2,\quad
    p(y_i\mid\theta^{(2)})=0.1,\quad
    p(y_i\mid\theta^{(3)})=0.05.
  \]

  \textbf{Which draw receives the greatest importance weight when we
    remove $y_i$? Why?}

  \uncover<2->{
    \medskip
    \textbf{Answer:} The third draw. The raw weights are
    \[
      r_i^{(s)}=\frac{1}{p(y_i\mid\theta^{(s)})}:
      \quad 5,\;10,\;20,
      \qquad\text{normalised: }\frac17,\;\frac27,\;\frac47.
    \]
    Observation $y_i$ penalised the third draw most strongly in the full
    posterior. Removing $y_i$ reverses that likelihood contribution.
  }
\end{frame}


\begin{frame}
\frametitle{LOO importance sampling: worked example}

  The three likelihood values are $0.2,\;0.1,\;0.05$.
  Their raw weights are $5,\;10,\;20$, with sum 35.
  \[
    \bar r_i^{(1)}=\frac17,\qquad
    \bar r_i^{(2)}=\frac27,\qquad
    \bar r_i^{(3)}=\frac47.
  \]
  \uncover<2->{
    The weighted predictive density is
    \begin{align*}
      \widehat p(y_i\mid y_{-i})
      &=\frac17(0.2)+\frac27(0.1)+\frac47(0.05)\\
      &=\frac{3}{35}\approx0.0857.
    \end{align*}
  }
  \uncover<3->{The unweighted training density is
    $(0.2+0.1+0.05)/3\approx0.1167$.
    Removing $y_i$ increases the relative weight of draws that predicted it poorly.}

  {\small Three draws illustrate the calculation; they are not enough for reliable estimation.}
\end{frame}

\begin{frame}

\frametitle{Recap: Pareto smoothed importance sampling}

  \begin{itemize}
  \item PSIS fits a generalized Pareto distribution to the largest raw weights
    and smooths these extreme weights.
  \item The estimated tail shape $\widehat k$ measures how heavy that tail is:
    larger values mean a few draws can dominate the estimate.
  \item<2-> As a rule of thumb, $\widehat k<0.7$ is reassuring;
    smaller values generally indicate more stable importance sampling.
  \item<3-> Software may use a stricter threshold when fewer posterior draws
    are available. Check its warnings.
  \item<4-> $\widehat k>0.7$ warns that the \uured{PSIS approximation is unreliable}.
    It does not mean that exact leave-one-out prediction is impossible.
  \end{itemize}
\end{frame}

  \begin{itemize}

    Reliable diagnostics for B do not establish that B predicts better.

\begin{frame}
\frametitle{What if $\widehat k$ is large?}

  \begin{enumerate}
  \item<1-> Inspect the influential observations and model assumptions.
    Check for data errors and whether the model represents plausible variation.
  \item<2-> For a few problematic observations, refit the model without
    each one and compute its LOO predictive density directly.
  \item<3-> For many problematic observations, use suitable K-fold validation
    or revise the model and evaluate it again.
  \end{enumerate}

  \uncover<4->{\uured{Large $\widehat k$ is a warning about the approximation,
    not an instruction to delete an observation.}
    Match the validation split to the prediction task.}

\subsection{K-fold CV}
\frame{\subsectionpage}

\begin{frame}{}

\frametitle{K-fold cross-validation}

\begin{itemize}
\item Split observations into $K$ folds $I_1,\ldots,I_K$, each containing
  approximately $1/K$ of the observations.
\pause
\begin{enumerate}
\item For $k=1,\ldots,K$:
  \begin{enumerate}
  \item hold out $y_{I_k}$; use $y_{-I_k}$ as the training data
  \item fit $p(\theta\mid y_{-I_k})$
  \item score only the held-out observations:
    \[
      L_k=\sum_{i\in I_k}\log\int p(y_i\mid\theta)p(\theta\mid y_{-I_k})d\theta
    \]
  \end{enumerate}
\item sum the scores: $\text{elpd}_{K\text{-fold}}=\sum_{k=1}^K L_k$
\end{enumerate}
\pause
\item Fit $K$ models, each with fewer training observations than in LOO
  when $K<n$.
\end{itemize}

\end{frame}

\begin{frame}{}

\frametitle{K-fold cross-validation}

\begin{enumerate}
\item K-fold cross-validation can approximate LOO
  \begin{enumerate}
    \item can be used all uses for LOO
  \end{enumerate}
\item K-fold cross-validation can be used for hierarchical models
  \begin{enumerate}
    \item good for leave-one-group-out
  \end{enumerate}
\item K-fold cross-validation can be used for time series
  \begin{enumerate}
    \item with leave-block-out
  \end{enumerate}
\end{enumerate}

\end{frame}

\subsection{Model Comparison}
\frame{\subsectionpage}

\begin{frame}
\frametitle{Paired model comparison}

  Score both models on the \uured{same held-out observations}:
  \[
    \ell_{mi}=\log p_m(y_i\mid y_{-i}),\qquad
    d_i=\ell_{1i}-\ell_{2i}.
  \]
  \[
    \widehat\Delta=\sum_{i=1}^n d_i
      =\mathrm{elpd}_{\mathrm{loo},1}-\mathrm{elpd}_{\mathrm{loo},2}.
  \]
  \begin{itemize}
    \item<2-> Positive differences favour model 1; negative differences favour model 2.
    \item<3-> Use the same prediction task, outcome units and validation split.
      Pairing accounts for observations that are easy or hard for both models.
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Uncertainty in a model comparison}

  For approximately independent held-out units,
  \[
    \bar d=\frac1n\sum_i d_i,\quad
    s_d^2=\frac{1}{n-1}\sum_i(d_i-\bar d)^2,
    \quad
    \mathrm{SE}(\widehat\Delta)\approx\sqrt{n\,s_d^2}.
  \]
  \begin{itemize}
  \item<2-> Use the variability of the \uured{paired differences}.
    Do not subtract the two models' standard errors.
  \item<3-> This is an approximation. Small samples, influential observations
    or dependent validation units require care.
  \item<4-> \texttt{loo\_compare()} reports each model minus the model with
    the largest ELPD. The best row is 0; other rows are negative.
    \texttt{se\_diff} is the paired standard error.
  \end{itemize}
  {\scriptsize \href{https://mc-stan.org/loo/reference/loo_compare.html}{loo: model comparison}.}
\end{frame}

      \item the difference between models is clear
    \end{enumerate}
  \item<2-> Do not use cross-validation to choose from a large set of models
    \begin{enumerate}
    \item selection process leads to overfitting
    % \item you may use projection predictive approach
    % \item useful when correlating variables make the posterior
    %   distribution analysis difficult\\
    %   {\small video, refs and demos  at \url{avehtari.github.io/modelselection/}\\
    %   and \href{http://link.springer.com/article/10.1007/s11222-016-9649-y}{Piironen \& Vehtari (2017)}}
    \end{enumerate}
  \item<3-> Overfitting in selection process is not unique for cross-validation
  \end{enumerate}
\end{frame}

\begin{frame}



\frametitle{Selection induced bias and overfitting}

  \begin{itemize}
  \item Selection induced bias in cross-validation
    \begin{itemize}
    \item same data is used to assess the performance and make the selection
    \item the selected model fits more to the data
    \item the CV estimate for the selected model is biased
%    \item recognized already, e.g., by Stone (1974)
    \end{itemize}
    \pause
  \item Performance of the selection process itself can be assessed
    using two level cross-validation, but it does not help choosing
    better models
    \pause
  \item Bigger problem if there is a large number of models as in
    covariate selection
  \end{itemize}

\end{frame}

\subsection{Information criteria}

\begin{frame}{}

\frametitle{Akaike information criterion}

\begin{enumerate}
  \item Recall the gap between training lppd and LOO performance
  \[
    \text{elpd}_\text{loo} = \sum_i \log p(y_i|y) - p_\text{loo}
  \]
  \pause
  \item Akaike information criterion
  \[
    \text{elpd}_\text{AIC} = \sum_i \log p(y_i|\hat{\theta}_\text{MLE}) - k
  \]
  \item AIC
  \[
    \text{AIC} = - 2 \text{elpd}_\text{AIC}
  \]
\end{enumerate}

\end{frame}


\begin{frame}{}

\frametitle{Deviance information criterion}

\begin{enumerate}
  \item Deviance information criterion
  \[
    \text{elpd}_\text{DIC} = \sum_i \log p(y_i|\hat{\theta}_\text{Bayes}) - p_\text{DIC}
  \]
  \item We estimate the effective number of parameters with
  \[
  p_\text{DIC1} = 2 (\log p(y|\hat{\theta}_\text{Bayes}) - \E_{p(\theta|y)} (\log p(y|\theta)) )
  \]
  or
  \[
  p_\text{DIC2} = 2 V_{p(\theta|y)} (\log p(y|\theta))
  \]
  \item DIC
  \[
    \text{DIC} = - 2 \text{elpd}_\text{DIC}
  \]
\end{enumerate}

\end{frame}

\begin{frame}{}

\frametitle{Widely applicable information criterion (WAIC)}

\begin{enumerate}
  \item Watanabe information criterion
  \[
    \text{elpd}_\text{WAIC} = \sum_i \log \uured{p(y_i|y)} - p_\text{WAIC}
  \]
  \item We estimate the effective number of parameters with
  \[
  p_\text{WAIC1} = 2 \sum^n_i (\log \E_{p(\theta|y)}(p(y_i|\theta)) - \E_{p(\theta|y)} (\log p(y_i|\theta)))
  \]
  or
  \[
  p_\text{WAIC2} = \sum_{i=1}^n \operatorname{Var}_{\theta\mid y}[\log p(y_i\mid\theta)]
  \]
  \item WAIC
  \[
    \text{WAIC} = - 2 \text{elpd}_\text{WAIC}
  \]
  \item WAIC can be seen as an approximation of the $elpd_\text{loo}$
\end{enumerate}

\end{frame}



\begin{frame}{}

\frametitle{WAIC vs PSIS-LOO}

\begin{enumerate}
  \item<2-> WAIC and LOO estimate pointwise predictive performance;
    they can differ substantially with finite data.
  \item<3-> PSIS-LOO is generally preferable when its importance-sampling
    diagnostics are satisfactory.
  \item<4-> Pareto $\widehat k$ helps diagnose unreliable PSIS approximations.
    Switching to WAIC is not a reliable fix for large $\widehat k$.
  \item<5-> LOO makes the held-out prediction task explicit;
    another validation split may be needed for a different task.
  \item<6-> ELPD: higher is better. On the $-2\times\text{ELPD}$ scale,
    lower is better; the ranking contains the same information.
\end{enumerate}

\end{frame}
